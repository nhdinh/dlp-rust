```rust
/// Removes and returns every buffered snapshot across all DLL keys.
///
/// Used by the periodic server-push task so each snapshot is forwarded exactly once
/// (the server store is append-only and does not deduplicate). Collects keys first,
/// then `remove`s each, so no `DashMap` guard is held across a remove and none is
/// returned to the caller (guards are not `Send` across `.await`).
///
/// # Returns
/// All previously buffered snapshots, sorted by QPC timestamp descending (newest first).
#[must_use]
pub fn drain_all(&self) -> Vec<DiagnosticSnapshot> {
    let keys: Vec<String> = self.snapshots.iter().map(|e| e.key().clone()).collect();
    let total: usize = keys
        .iter()
        .map(|k| self.snapshots.get(k).map(|v| v.len()).unwrap_or(0))
        .sum();
    let mut out: Vec<DiagnosticSnapshot> = Vec::with_capacity(total);
    for key in keys {
        if let Some((_, mut v)) = self.snapshots.remove(&key) {
            out.append(&mut v);
        }
    }
    out.sort_by(|a, b| b.timestamp_qpc.cmp(&a.timestamp_qpc));
    out
}
```

const DIAGNOSTIC_PUSH_INTERVAL: Duration = Duration::from_secs(60);

async fn diagnostic_push_loop(
    server_client: crate::server_client::ServerClient,
    diagnostic_aggregator: Arc<crate::diagnostic_aggregator::DiagnosticAggregator>,
    agent_id: String,
    interval: Duration,
    mut shutdown: tokio::sync::watch::Receiver<bool>,
) {
    let mut interval = tokio::time::interval(interval);
    loop {
        tokio::select! {
            _ = interval.tick() => {
                let snapshots = diagnostic_aggregator.drain_all(); // owned Vec; guards dropped before await
                if snapshots.is_empty() {
                    debug!("no diagnostic snapshots to push");
                    continue;
                }
                if let Err(e) = server_client
                    .submit_diagnostic_snapshot(&agent_id, 0, &snapshots) // pid 0 = aggregate batch (A3)
                    .await
                {
                    warn!(error = %e, count = snapshots.len(), "failed to submit diagnostics");
                }
            }
            _ = shutdown.changed() => {
                // Flush-and-stop: forward whatever is buffered before exiting.
                let snapshots = diagnostic_aggregator.drain_all();
                if !snapshots.is_empty() {
                    let _ = server_client.submit_diagnostic_snapshot(&agent_id, 0, &snapshots).await;
                }
                debug!("diagnostic push loop shutting down");
                return;
            }
        }
    }
}
