<objective>
Implement mount-time blocking for unregistered fixed disks. When a disk arrives, check the frozen allowlist (instance_id_map) BEFORE adding it to drive_letter_map. If unregistered, remove the drive letter via DefineDosDeviceW and take the volume offline via IOCTL_VOLUME_OFFLINE. Emit an audit event. Keep I/O-time blocking as fallback.

Purpose: Move disk enforcement earlier in the pipeline — prevent the disk from appearing in Explorer rather than blocking writes after the fact.
Output: Mount-time blocking integrated into disk arrival handling with audit events.
</objective>

#[cfg(windows)]
fn block_disk_at_mount_time(
    letter: char,
    disk: &DiskIdentity,
    audit_ctx: &EmitContext,
) -> Result<(), String> {
    // Step 1: Remove drive letter from DOS namespace
    let drive_str = format!("{}:", letter);
    let wide: Vec<u16> = drive_str.encode_utf16().chain(std::iter::once(0)).collect();
    let success = unsafe {
        windows::Win32::Storage::FileSystem::DefineDosDeviceW(
            windows::Win32::Storage::FileSystem::DDD_REMOVE_DEFINITION,
            windows::core::PCWSTR(wide.as_ptr()),
            windows::core::PCWSTR::null(),
        )
    };
    if !success.as_bool() {
        let err = windows::core::Error::from_win32();
        warn!(letter = %letter, error = %err, "DefineDosDeviceW failed to remove drive letter");
    } else {
        info!(letter = %letter, instance_id = %disk.instance_id, "Drive letter removed for unregistered disk");
    }

    // Step 2: Defense-in-depth — dismount and offline the volume
    let volume_path = format!(r"\\.\{}:", letter);
    let wide_path: Vec<u16> = volume_path.encode_utf16().chain(std::iter::once(0)).collect();
    unsafe {
        let handle = windows::Win32::Storage::FileSystem::CreateFileW(
            windows::core::PCWSTR(wide_path.as_ptr()),
            windows::Win32::Storage::FileSystem::FILE_GENERIC_READ.0 | windows::Win32::Storage::FileSystem::FILE_GENERIC_WRITE.0,
            windows::Win32::Storage::FileSystem::FILE_SHARE_READ | windows::Win32::Storage::FileSystem::FILE_SHARE_WRITE,
            None,
            windows::Win32::Storage::FileSystem::OPEN_EXISTING,
            windows::Win32::Storage::FileSystem::FILE_FLAGS_AND_ATTRIBUTES(0),
            None,
        );
        if handle != windows::Win32::Foundation::INVALID_HANDLE_VALUE {
            let mut bytes_returned = 0u32;
            // Dismount first
            let _ = windows::Win32::System::Ioctl::DeviceIoControl(
                handle,
                windows::Win32::System::Ioctl::FSCTL_DISMOUNT_VOLUME,
                None,
                0,
                None,
                0,
                Some(&mut bytes_returned),
                None,
            );
            // Then offline
            let _ = windows::Win32::System::Ioctl::DeviceIoControl(
                handle,
                windows::Win32::System::Ioctl::IOCTL_VOLUME_OFFLINE,
                None,
                0,
                None,
                0,
                Some(&mut bytes_returned),
                None,
            );
            let _ = windows::Win32::Foundation::CloseHandle(handle);
        }
    }

    // Step 3: Emit audit event
    emit_disk_mount_blocked(disk, audit_ctx);

    Ok(())
}
```

fn emit_disk_mount_blocked(disk: &DiskIdentity, audit_ctx: &EmitContext) {
    let event = dlp_common::AuditEvent {
        event_type: dlp_common::EventType::DiskMountBlocked,
        timestamp: chrono::Utc::now(),
        details: serde_json::json!({
            "instance_id": disk.instance_id,
            "bus_type": format!("{:?}", disk.bus_type),
            "model": disk.model,
            "drive_letter": disk.drive_letter.map(|c| c.to_string()),
        }),
        ..Default::default()
    };
    audit_ctx.emit(event);
}
```

fn on_disk_arrival_inner(device_path: &str, live_disks: &[DiskIdentity], audit_ctx: &EmitContext) {
    let enumerator = get_disk_enumerator();
    let instance_id_map = enumerator.instance_id_map.read();
    
    for disk in live_disks {
        let Some(letter) = disk.drive_letter else { continue };
        
        // NEW: Check allowlist before adding to drive_letter_map
        if !instance_id_map.contains_key(&disk.instance_id) {
            // Unregistered disk — block at mount time
            if let Err(e) = block_disk_at_mount_time(letter, disk, audit_ctx) {
                warn!(letter = %letter, error = %e, "Mount-time block failed, falling back to I/O-time blocking");
            }
            // Do NOT insert into drive_letter_map — keeps disk invisible
            continue;
        }
        
        // Existing: insert registered disks into drive_letter_map
        {
            let mut map = enumerator.drive_letter_map.write();
            if map.contains_key(&letter) { continue; }
            map.insert(letter, disk.clone());
        }
    }
}
