   ```rust
   impl EngineClient {
       pub async fn list_approvals(&self, status: Option<&str>, page: u32, per_page: u32) -> Result<ApprovalListResponse, ClientError> {
           let mut url = format!("{}/admin/approvals?page={}&per_page={}", self.base_url, page, per_page);
           if let Some(s) = status {
               url.push_str(&format!("&status={}", urlencoding::encode(s)));
           }
           let resp = self.client.get(&url).bearer_auth(&self.token).send().await?;
           Self::handle_response(resp).await
       }

       pub async fn get_approval(&self, id: &str) -> Result<ApprovalDetailResponse, ClientError> {
           let url = format!("{}/admin/approvals/{}", self.base_url, id);
           let resp = self.client.get(&url).bearer_auth(&self.token).send().await?;
           Self::handle_response(resp).await
       }

       pub async fn grant_approval(&self, id: &str, valid_until: &str, signature: Option<&str>) -> Result<GrantResponse, ClientError> {
           let url = format!("{}/admin/approvals/{}/grant", self.base_url, id);
           let body = serde_json::json!({
               "valid_until": valid_until,
               "signature": signature,
           });
           let resp = self.client.post(&url).bearer_auth(&self.token).json(&body).send().await?;
           Self::handle_response(resp).await
       }

       pub async fn reject_approval(&self, id: &str, reason: Option<&str>) -> Result<ApprovalResponse, ClientError> {
           let url = format!("{}/admin/approvals/{}/reject", self.base_url, id);
           let body = serde_json::json!({ "reason": reason });
           let resp = self.client.post(&url).bearer_auth(&self.token).json(&body).send().await?;
           Self::handle_response(resp).await
       }

       pub async fn revoke_approval(&self, id: &str) -> Result<ApprovalResponse, ClientError> {
           let url = format!("{}/admin/approvals/{}/revoke", self.base_url, id);
           let resp = self.client.post(&url).bearer_auth(&self.token).send().await?;
           Self::handle_response(resp).await
       }
   }
   ```

   ```rust
   pub fn draw_approval_list(f: &mut Frame, app: &App, area: Rect) {
       let state = match &app.screen {
           Screen::ApprovalList => &app.approval_list_state,
           _ => return,
       };
       
       let total_pages = ((state.total as f64) / (state.per_page as f64)).ceil() as u32;
       let total_pages = total_pages.max(1);
       
       let header = if let Some(filter_str) = state.filter.as_str() {
           format!("Approval Management [{}] — Page {} of {} ({} total)", 
               filter_str, state.page, total_pages, state.total)
       } else {
           format!("Approval Management — Page {} of {} ({} total)", 
               state.page, total_pages, state.total)
       };
       
       // Table columns: Requester | Object | Action | Status | Expires
       let rows: Vec<Row> = state.approvals.iter().enumerate().map(|(i, resp)| {
           let approval = &resp.approval;
           let status_style = match approval.status {
               ApprovalStatus::Pending => Style::default().fg(Color::Yellow),
               ApprovalStatus::Approved => Style::default().fg(Color::Green),
               ApprovalStatus::Rejected => Style::default().fg(Color::Red),
               ApprovalStatus::Revoked => Style::default().fg(Color::DarkGray),
               ApprovalStatus::Expired => Style::default().fg(Color::DarkGray).add_modifier(Modifier::CROSSED_OUT),
           };
           let t4_badge = if resp.tier.as_deref() == Some("T4") { " [T4]" } else { "" };
           let status_text = format!("{}{}", approval.status.to_string(), t4_badge);
           let cells = vec![
               Cell::from(approval.requester_sid.clone()),
               Cell::from(approval.data_object_id.clone()),
               Cell::from(approval.allowed_action.clone()),
               Cell::from(status_text).style(status_style),
               Cell::from(approval.valid_until.clone().unwrap_or_else(|| "—".to_string())),
           ];
           let style = if i == state.selected {
               Style::default().add_modifier(Modifier::REVERSED)
           } else {
               Style::default()
           };
           Row::new(cells).style(style)
       }).collect();
       
       let table = Table::new(rows, vec![
           Constraint::Percentage(20),
           Constraint::Percentage(25),
           Constraint::Percentage(15),
           Constraint::Percentage(20),
           Constraint::Percentage(20),
       ])
       .header(Row::new(vec!["Requester", "Object", "Action", "Status", "Expires"])
           .style(Style::default().add_modifier(Modifier::BOLD)))
       .block(Block::default().borders(Borders::ALL).title(header));
       
       f.render_widget(table, area);
       
       // Footer hint
       let hint = Paragraph::new(APPROVAL_LIST_HINTS)
           .style(Style::default().fg(Color::DarkGray));
       f.render_widget(hint, Rect { height: 1, ..area });
   }
   ```

   ```rust
   pub async fn handle_approval_list(app: &mut App, key: KeyEvent) -> Result<Option<Screen>, ClientError> {
       let state = &mut app.approval_list_state;
       
       match key.code {
           KeyCode::Up => {
               if state.selected > 0 {
                   state.selected -= 1;
               }
           }
           KeyCode::Down => {
               if state.selected + 1 < state.approvals.len() {
                   state.selected += 1;
               }
           }
           KeyCode::Char('g') => {
               // Grant selected pending approval
               if let Some(resp) = state.approvals.get(state.selected) {
                   if resp.approval.status == ApprovalStatus::Pending {
                       return Ok(Some(Screen::ApprovalGrant {
                           approval_id: resp.approval.id.clone(),
                       }));
                   } else {
                       state.status_message = "Can only grant pending approvals".to_string();
                   }
               }
           }
   ```
