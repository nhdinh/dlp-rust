   ```rust
   impl EngineClient {
       /// GET /admin/approvals — list approvals with optional status filter.
       pub async fn list_approvals(&self, status: Option<&str>) -> Result<Vec<serde_json::Value>> {
           let path = match status {
               Some(s) => format!("/admin/approvals?status={}", s),
               None => "/admin/approvals".to_string(),
           };
           let resp = self.get(&path).await?;
           resp.as_array()
               .cloned()
               .ok_or_else(|| anyhow::anyhow!("expected array response"))
       }

       /// POST /admin/approvals/{id}/grant — grant a pending approval.
       pub async fn grant_approval(
           &self,
           id: &str,
           valid_until: &str,
           signature: Option<&str>,
       ) -> Result<serde_json::Value> {
           let body = serde_json::json!({
               "valid_until": valid_until,
               "signature": signature,
           });
           self.post(&format!("/admin/approvals/{}/grant", id), body).await
       }

       /// POST /admin/approvals/{id}/reject — reject a pending approval.
       pub async fn reject_approval(&self, id: &str, reason: Option<&str>) -> Result<serde_json::Value> {
           let body = serde_json::json!({
               "reason": reason,
           });
           self.post(&format!("/admin/approvals/{}/reject", id), body).await
       }

       /// POST /admin/approvals/{id}/revoke — revoke an approved approval.
       pub async fn revoke_approval(&self, id: &str) -> Result<serde_json::Value> {
           self.post(&format!("/admin/approvals/{}/revoke", id), serde_json::json!({})).await
       }

       /// GET /admin/approvals/{id} — get a single approval.
       pub async fn get_approval(&self, id: &str) -> Result<serde_json::Value> {
           self.get(&format!("/admin/approvals/{}", id)).await
       }
   }
   ```

      ```rust
      fn handle_approval_list(app: &mut App, key: KeyEvent) {
          let Screen::ApprovalList { approvals, selected, filter } = &mut app.screen else { return };
          match key.code {
              KeyCode::Up => {
                  if *selected > 0 { *selected -= 1; }
              }
              KeyCode::Down => {
                  if *selected + 1 < approvals.len() { *selected += 1; }
              }
              KeyCode::Char('g') => {
                  // Grant selected pending approval
                  if let Some(approval) = approvals.get(*selected) {
                      if approval["status"] == "pending" {
                          app.screen = Screen::ApprovalGrant {
                              approval: approval.clone(),
                              expiry: ExpiryOption::OneHour,
                              custom_expiry: String::new(),
                              t4_signature_confirmed: false,
                              selected_row: 0,
                          };
                      }
                  }
              }
              KeyCode::Char('r') => {
                  // Revoke selected approved approval
                  if let Some(approval) = approvals.get(*selected) {
                      if approval["status"] == "approved" {
                          let id = approval["id"].as_str().unwrap_or("").to_string();
                          app.screen = Screen::Confirm {
                              message: format!("Revoke approval {}?", id),
                              yes_selected: false,
                              purpose: ConfirmPurpose::RevokeApproval { id },
                          };
                      }
                  }
              }
              KeyCode::Char('v') => {
                  // View detail
                  if let Some(approval) = approvals.get(*selected) {
                      app.screen = Screen::ApprovalDetail {
                          approval: approval.clone(),
                      };
                  }
              }
              KeyCode::Char('f') => {
                  *filter = filter.next();
                  // Trigger reload
                  app.status = ("Reloading...".to_string(), StatusKind::Info);
                  // The event loop will reload on next tick
              }
              KeyCode::Esc => {
                  app.screen = Screen::SystemMenu { selected: 0 };
              }
              _ => {}
          }
      }
      ```

      ```rust
      fn handle_approval_grant(app: &mut App, key: KeyEvent) {
          let Screen::ApprovalGrant { approval, expiry, custom_expiry, t4_signature_confirmed, selected_row } = &mut app.screen else { return };
          match key.code {
              KeyCode::Up => {
                  if *selected_row > 0 { *selected_row -= 1; }
              }
              KeyCode::Down => {
                  if *selected_row < 2 { *selected_row += 1; }
              }
              KeyCode::Enter => {
                  if *selected_row == 2 {
                      // Submit grant
                      let id = approval["id"].as_str().unwrap_or("").to_string();
                      let valid_until = match expiry {
                          ExpiryOption::OneHour => (chrono::Utc::now() + chrono::Duration::hours(1)).to_rfc3339(),
                          ExpiryOption::FourHours => (chrono::Utc::now() + chrono::Duration::hours(4)).to_rfc3339(),
                          ExpiryOption::EightHours => (chrono::Utc::now() + chrono::Duration::hours(8)).to_rfc3339(),
                          ExpiryOption::TwentyFourHours => (chrono::Utc::now() + chrono::Duration::hours(24)).to_rfc3339(),
                          ExpiryOption::Custom => custom_expiry.clone(),
                      };
                      // Check T4 requirement
                      let is_t4 = approval["tier"].as_str() == Some("T4");
                      let signature = if is_t4 {
                          if !*t4_signature_confirmed {
                              app.status = ("T4 approval requires Board signature confirmation".to_string(), StatusKind::Error);
                              return;
                          }
                          // In Phase 61, the admin provides the signature hex via a text input that precedes this form.
                          // For now, the signature is collected via a pre-form text input and stored in app state.
                          // If not available, show error.
                          None // Will be populated from app state
                      } else {
                          None
                      };
                      // Async grant call
                      app.status = ("Granting approval...".to_string(), StatusKind::Info);
                      // The actual HTTP call is made by the event loop via app.pending_action
                      app.pending_action = Some(PendingAction::GrantApproval { id, valid_until, signature });
                  }
              }
              KeyCode::Esc => {
                  app.screen = Screen::ApprovalList {
                      approvals: Vec::new(),
                      selected: 0,
                      filter: ApprovalFilter::All,
                  };
              }
              KeyCode::Char(' ') => {
                  if *selected_row == 1 {
                      *t4_signature_confirmed = !*t4_signature_confirmed;
                  }
              }
              _ => {}
          }
      }
      ```

      ```rust
      fn draw_approval_list(
          frame: &mut Frame,
          area: Rect,
          approvals: &[serde_json::Value],
          selected: usize,
          filter: ApprovalFilter,
      ) {
          let header = match filter {
              ApprovalFilter::All => "Approval Management".to_string(),
              _ => format!("Approval Management [Filter: {}]", filter.as_str()),
          };
          let block = Block::default().title(header).borders(Borders::ALL);
          let inner = block.inner(area);
          frame.render_widget(block, area);

          if approvals.is_empty() {
              let text = Paragraph::new("No approvals found.");
              frame.render_widget(text, inner);
              return;
          }

          let header_row = Row::new(vec!["Requester", "Object", "Action", "Status", "Expires"]);
          let rows: Vec<Row> = approvals.iter().enumerate().map(|(i, a)| {
              let status = a["status"].as_str().unwrap_or("");
              let status_color = match status {
                  "pending" => Color::Yellow,
                  "approved" => Color::Green,
                  "rejected" => Color::Red,
                  "revoked" | "expired" => Color::DarkGray,
                  _ => Color::White,
              };
              let mut style = Style::default().fg(status_color);
              if status == "expired" {
                  style = style.add_modifier(Modifier::CROSSED_OUT);
              }
              if i == selected {
                  style = style.add_modifier(Modifier::REVERSED);
              }
              let expires = a["valid_until"].as_str().unwrap_or("-");
              Row::new(vec![
                  a["requester_sid"].as_str().unwrap_or("").to_string(),
                  a["data_object_id"].as_str().unwrap_or("").to_string(),
                  a["allowed_action"].as_str().unwrap_or("").to_string(),
                  status.to_string(),
                  expires.to_string(),
              ]).style(style)
          }).collect();

          let table = Table::new(rows, vec![
              Constraint::Percentage(25),
              Constraint::Percentage(25),
              Constraint::Percentage(15),
              Constraint::Percentage(15),
              Constraint::Percentage(20),
          ])
          .header(header_row)
          .column_spacing(1);
          frame.render_widget(table, inner);

          // Footer hint
          let hint = Paragraph::new("[g] Grant  [r] Revoke  [v] View  [f] Filter  [Esc] Back");
          frame.render_widget(hint, inner);
      }
      ```
