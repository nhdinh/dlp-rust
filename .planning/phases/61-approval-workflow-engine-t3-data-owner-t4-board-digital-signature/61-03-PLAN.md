      impl ApprovalCache {
          pub fn new() -> Self {
              Self { cache: DashMap::new(), verifying_key: None }
          }

          /// Set the server's Ed25519 public key for JWT verification.
          pub fn set_public_key(&mut self, pubkey_hex: &str) -> Result<(), AgentError> {
              let bytes = hex::decode(pubkey_hex)?;
              let key = ed25519_dalek::VerifyingKey::from_bytes(&bytes)?;
              self.verifying_key = Some(key);
              Ok(())
          }

          /// Cache an approval token. TTL derived from claims.exp.
          pub fn insert(&self, key: ApprovalCacheKey, token: String, claims: ApprovalClaims) {
              let expires_at = chrono::DateTime::from_timestamp(claims.exp, 0)
                  .unwrap_or_else(|| Utc::now() + chrono::Duration::hours(1));
              self.cache.insert(key.encode(), CachedApproval { token, claims, expires_at });
          }

          /// Check approval cache. Re-verifies JWT signature using cached public key.
          /// Returns Some(ALLOW) if valid, None otherwise.
          /// PERFORMANCE NOTE: Ed25519 verification is ~50us. For high-frequency hooks,
          /// consider caching verified claims with periodic re-verification.
          pub fn check(&self, key: &ApprovalCacheKey, request_dst: Option<&str>) -> Option<EvaluateResponse> {
              let entry = self.cache.get(&key.encode())?;
              if Utc::now() > entry.expires_at {
                  drop(entry);
                  self.cache.remove(&key.encode());
                  return None;
              }
              // Re-verify JWT signature using cached public key
              if let Some(ref vk) = self.verifying_key {
                  let dec_key = jsonwebtoken::DecodingKey::from_ed_der(&vk.to_bytes());
                  let mut validation = jsonwebtoken::Validation::new(jsonwebtoken::Algorithm::EdDSA);
                  validation.set_issuer(&["dlp-server"]);
                  if jsonwebtoken::decode::<ApprovalClaims>(&entry.token, &dec_key, &validation).is_err() {
                      tracing::warn!("approval cache entry failed signature re-verification");
                      return None;
                  }
              }
              // Validate destination scope
              if let Some(ref approved_dst) = entry.claims.dst {
                  if !approved_dst.is_empty() {
                      if let Some(req_dst) = request_dst {
                          if !scope_matches(req_dst, approved_dst) {
                              return None;
                          }
                      }
                  }
              }
              Some(EvaluateResponse {
                  decision: Decision::ALLOW,
                  matched_policy_id: Some(format!("approval:{}", entry.claims.jti)),
                  reason: "approved via override token".to_string(),
              })
          }

          pub fn remove(&self, key: &ApprovalCacheKey) {
              self.cache.remove(&key.encode());
          }

          pub fn sweep_expired(&self) {
              let now = Utc::now();
              let to_remove: Vec<String> = self.cache.iter()
                  .filter(|e| now > e.expires_at)
                  .map(|e| e.key().clone())
                  .collect();
              for key in to_remove {
                  self.cache.remove(&key);
              }
          }
      }

      fn scope_matches(request_dst: &str, approved_dst: &str) -> bool {
          if approved_dst.is_empty() || approved_dst == "*" { return true; }
          if request_dst == approved_dst { return true; }
          if approved_dst.ends_with(":*") {
              let prefix = &approved_dst[..approved_dst.len() - 2];
              return request_dst.starts_with(prefix) && request_dst.len() > prefix.len();
          }
          false
      }
      ```

   use dlp_common::approval::{ApprovalCacheKey, ApprovalClaims};

   /// Fetches the server's Ed25519 public key for offline JWT verification.
   /// Called once at startup; cached in ApprovalCache.
   pub async fn fetch_public_key(
       client: &reqwest::Client,
       server_url: &str,
   ) -> Result<String, anyhow::Error> {
       let resp = client
           .get(format!("{}/agent/approvals/public-key", server_url))
           .send()
           .await?;
       if resp.status().is_success() {
           let key_hex = resp.text().await?;
           // Validate: must be 64 hex chars = 32 bytes
           if key_hex.len() != 64 || !key_hex.chars().all(|c| c.is_ascii_hexdigit()) {
               return Err(anyhow::anyhow!("invalid public key format: expected 64 hex chars"));
           }
           Ok(key_hex)
       } else {
           Err(anyhow::anyhow!("server returned {}: {}", resp.status(), resp.text().await?))
       }
   }

   /// Submits an approval request to the server via HTTP.
   pub async fn submit_approval_request(
       client: &reqwest::Client,
       server_url: &str,
       request: &ApprovalRequest,
   ) -> Result<String, anyhow::Error> {
       let resp = client
           .post(format!("{}/agent/approval-request", server_url))
           .json(request)
           .send()
           .await?;
       if resp.status().is_success() {
           let body: serde_json::Value = resp.json().await?;
           Ok(body["id"].as_str().unwrap_or("").to_string())
       } else {
           Err(anyhow::anyhow!("server returned {}: {}", resp.status(), resp.text().await?))
       }
   }

   /// Syncs all active approvals from the server on startup, reconnection, and periodic poll.
   /// Calls GET /agent/approvals/active and reconciles with local cache.
   /// REVOCATION PROPAGATION: Approvals not in the server response are removed from cache.
   pub async fn sync_active_approvals(
       client: &reqwest::Client,
       server_url: &str,
       approval_cache: &ApprovalCache,
   ) -> Result<usize, anyhow::Error> {
       let resp = client
           .get(format!("{}/agent/approvals/active", server_url))
           .send()
           .await?;
       if !resp.status().is_success() {
           return Err(anyhow::anyhow!("server returned {}: {}", resp.status(), resp.text().await?));
       }
       let active: Vec<ActiveApprovalResponse> = resp.json().await?;
       let count = active.len();

       // Build set of active keys for revocation detection
       let mut active_keys = std::collections::HashSet::new();
       for item in &active {
           let key = ApprovalCacheKey {
               sid: item.claims.sub.clone(),
               obj_id: item.claims.obj.clone(),
               action: item.claims.act.clone(),
               dst: item.claims.dst.clone(),
           };
           active_keys.insert(key.encode());
           approval_cache.insert(key, item.token.clone(), item.claims.clone());
       }

       // Remove cache entries that are no longer active (revoked or expired server-side)
       let to_remove: Vec<String> = approval_cache.cache.iter()
           .filter(|e| !active_keys.contains(e.key()))
           .map(|e| e.key().clone())
           .collect();
       for key in to_remove {
           approval_cache.cache.remove(&key);
           tracing::info!("removed revoked/expired approval from cache: {}", key);
       }

       tracing::info!(count, "synced active approvals from server");
       Ok(count)
   }
   ```
