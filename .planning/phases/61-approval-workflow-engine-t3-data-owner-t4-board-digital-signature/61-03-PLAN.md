      ```rust
      /// Cache an approval token for the given key. TTL is derived from claims.exp.
      pub fn cache_approval(&self, key: String, token: String, claims: ApprovalClaims) {
          let expires_at = chrono::DateTime::from_timestamp(claims.exp, 0)
              .unwrap_or_else(|| Utc::now() + chrono::Duration::hours(1));
          self.approval_cache.insert(key, CachedApproval { token, claims, expires_at });
          tracing::info!(key, "approval cached");
      }

      /// Check if an approval exists and is not expired for the given key.
      /// Re-verifies JWT signature before returning ALLOW.
      /// Validates destination scope matches the request's scope.
      /// Returns Some(EvaluateResponse::ALLOW) if valid, None otherwise.
      pub fn check_approval(&self, key: &str, request_dst: Option<&str>) -> Option<EvaluateResponse> {
          let entry = self.approval_cache.get(key)?;
          if Utc::now() > entry.expires_at {
              drop(entry);
              self.approval_cache.remove(key);
              tracing::info!(key, "approval expired, removed from cache");
              return None;
          }
          // Re-verify JWT signature to prevent cache poisoning
          if let Some(ref token_service) = self.approval_token_service {
              if token_service.verify_token(&entry.token).is_err() {
                  tracing::warn!(key, "approval cache entry failed signature re-verification");
                  return None;
              }
          }
          // Validate destination scope: request_dst must be within the approved scope
          // Approved scope of "None" or "" means "any destination" (wildcard)
          // Approved scope of "USB:*" matches "USB:DRIVE_E"
          // Approved scope of "USB:DRIVE_E" only matches "USB:DRIVE_E"
          if let Some(ref approved_dst) = entry.claims.dst {
              if !approved_dst.is_empty() {
                  if let Some(req_dst) = request_dst {
                      if !scope_matches(req_dst, approved_dst) {
                          tracing::info!(key, request_dst = req_dst, approved_dst, "approval destination scope mismatch");
                          return None;
                      }
                  }
              }
          }
          Some(EvaluateResponse {
              decision: Decision::ALLOW,
              matched_policy_id: Some(format!("approval:{}", entry.claims.jti)),
              reason: "approved via override token".to_string(),
          })
      }

      /// Remove an approval from the cache (e.g., on revocation).
      pub fn invalidate_approval(&self, key: &str) {
          if self.approval_cache.remove(key).is_some() {
              tracing::info!(key, "approval invalidated");
          }
      }

      /// Remove all expired approvals from the cache.
      pub fn sweep_expired_approvals(&self) {
          let now = Utc::now();
          let to_remove: Vec<String> = self.approval_cache
              .iter()
              .filter(|e| now > e.expires_at)
              .map(|e| e.key().clone())
              .collect();
          for key in to_remove {
              self.approval_cache.remove(&key);
              tracing::info!(key, "swept expired approval");
          }
      }
      ```
   d. Add scope matching helper:
      ```rust
      /// Check if a request destination matches an approved destination scope.
      /// Supports hierarchical wildcards: "USB:*" matches "USB:DRIVE_E" and "USB:DRIVE_F".
      /// Exact match always succeeds. Empty approved scope means "any".
      fn scope_matches(request_dst: &str, approved_dst: &str) -> bool {
          if approved_dst.is_empty() || approved_dst == "*" {
              return true;
          }
          if request_dst == approved_dst {
              return true;
          }
          // Hierarchical wildcard: "USB:*" matches "USB:DRIVE_E"
          if approved_dst.ends_with(":*") {
              let prefix = &approved_dst[..approved_dst.len() - 2];
              return request_dst.starts_with(prefix) && request_dst.len() > prefix.len();
          }
          false
      }
      ```

   ```rust
   /// Three-stage ABAC evaluation pipeline:
   /// 1. NTFS check — coarse-grained access control (existing)
   /// 2. ABAC policy check — fine-grained dynamic control (existing)
   /// 3. Approval cache override — only applies when NTFS=ALLOW and ABAC=DENY
   ///
   /// Critical invariant: If NTFS ALLOW and ABAC DENY -> check approval cache.
   /// If approval cache hits and is valid -> ALLOW.
   /// If approval cache misses -> FINAL RESULT = DENY.
   pub fn evaluate(&self, ctx: &AbacContext, label_service: Option<&LabelService>) -> EvaluateResponse {
       // Stage 1: NTFS check (existing logic)
       // ... existing NTFS evaluation ...
       
       // Stage 2: ABAC policy check (existing logic)
       let policy_result = self.evaluate_policies(ctx);
       
       // Stage 3: Approval cache override
       // Only check approval cache if NTFS=ALLOW and ABAC=DENY
       if policy_result.decision == Decision::DENY {
           // Resolve resource path to data_object_id for cache lookup
           let data_object_id = if let Some(ref ls) = label_service {
               ls.resolve_path_to_id(&ctx.resource.path)
                   .unwrap_or_else(|| ctx.resource.path.clone())
           } else {
               ctx.resource.path.clone()
           };
           
           // Build cache key including destination scope
           let cache_key = approval_cache_key(
               &ctx.subject.user_sid,
               &data_object_id,
               &format!("{:?}", ctx.action),
               ctx.resource.destination_scope.as_deref()
           );
           
           if let Some(approval_resp) = self.check_approval(&cache_key, ctx.resource.destination_scope.as_deref()) {
               return approval_resp;
           }
       }
       
       // Default: return policy result (ALLOW or DENY)
       policy_result
   }
   ```

   ```rust
   /// Submits an approval request to the server via HTTP.
   pub async fn submit_approval_request(
       client: &reqwest::Client,

           .send()
           .await?;
       if resp.status().is_success() {
           let body: serde_json::Value = resp.json().await?;
           Ok(body["id"].as_str().unwrap_or("").to_string())
       } else {
           Err(anyhow::anyhow!("server returned {}: {}", resp.status(), resp.text().await?))
       }
   }

   /// Receives and caches an approval token from the server.
   pub async fn receive_approval_token(
       client: &reqwest::Client,
       server_url: &str,
       token: &str,
   ) -> Result<(), anyhow::Error> {
       let resp = client
           .post(format!("{}/agent/approval-token", server_url))
           .json(&serde_json::json!({ "token": token }))
           .send()
           .await?;
       if resp.status().is_success() {
           Ok(())
       } else {
           Err(anyhow::anyhow!("server returned {}: {}", resp.status(), resp.text().await?))
       }
   }

   /// Syncs all active approvals from the server on startup or reconnection.
   /// Calls GET /agent/approvals/active and caches each returned token.
   pub async fn sync_active_approvals(
       client: &reqwest::Client,
       server_url: &str,
       policy_store: &PolicyStore,
   ) -> Result<usize, anyhow::Error> {
       let resp = client
           .get(format!("{}/agent/approvals/active", server_url))
           .send()
           .await?;
       if !resp.status().is_success() {
           return Err(anyhow::anyhow!("server returned {}: {}", resp.status(), resp.text().await?));
       }
       let active: Vec<ActiveApprovalResponse> = resp.json().await?;
       let count = active.len();
       for item in active {
           let cache_key = approval_cache_key(
               &item.claims.sub,
               &item.claims.obj,
               &item.claims.act,
               item.claims.dst.as_deref()
           );
           policy_store.cache_approval(cache_key, item.token, item.claims);
       }
       tracing::info!(count, "synced active approvals from server");
       Ok(count)
   }
   ```
