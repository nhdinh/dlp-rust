   ```rust
   use ed25519_dalek::{SigningKey, pkcs8};
   use jsonwebtoken::{encode, decode, Header, Algorithm, Validation, EncodingKey, DecodingKey};
   use serde::{Serialize, Deserialize};

   #[derive(Debug, Serialize, Deserialize)]
   struct Claims { sub: String, exp: usize, iat: usize, iss: String }

   fn main() {
       let signing_key = SigningKey::generate(&mut rand::thread_rng());
       let verifying_key = signing_key.verifying_key();

       // Test pkcs8 DER encoding
       let pkcs8_der = signing_key.to_pkcs8_der().unwrap();
       let der_bytes = pkcs8_der.as_bytes();
       println!("PKCS#8 DER length: {}", der_bytes.len());

       // Test jsonwebtoken EncodingKey::from_ed_der
       let enc_key = EncodingKey::from_ed_der(der_bytes);

       // Sign a token
       let claims = Claims { sub: "test".to_string(), exp: 2000000000, iat: 1000000000, iss: "dlp-server".to_string() };
       let token = encode(&Header::new(Algorithm::EdDSA), &claims, &enc_key).unwrap();
       println!("Token: {}", token);

       // Verify with DecodingKey::from_ed_der
       let dec_key = DecodingKey::from_ed_der(&verifying_key.to_bytes());
       let mut validation = Validation::new(Algorithm::EdDSA);
       validation.set_issuer(&["dlp-server"]);
       let decoded = decode::<Claims>(&token, &dec_key, &validation).unwrap();
       println!("Decoded: {:?}", decoded.claims);

       // Test wrong issuer rejection
       let bad_claims = Claims { sub: "test".to_string(), exp: 2000000000, iat: 1000000000, iss: "attacker".to_string() };
       let bad_token = encode(&Header::new(Algorithm::EdDSA), &bad_claims, &enc_key).unwrap();
       let bad_result = decode::<Claims>(&bad_token, &dec_key, &validation);
       assert!(bad_result.is_err(), "Wrong issuer should be rejected");
       println!("Wrong issuer correctly rejected");
   }
   ```
