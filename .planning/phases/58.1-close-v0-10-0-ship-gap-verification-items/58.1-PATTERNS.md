```rust
pub fn send_raw_oneway(pipe_name: &str, payload: &[u8]) -> Result<(), PipeError> {
    const CONNECT_TIMEOUT_MS: u32 = 50;
    let pipe = connect_pipe(pipe_name, CONNECT_TIMEOUT_MS)?;
    unsafe {
        let mode = PIPE_READMODE_MESSAGE;
        if let Err(e) = SetNamedPipeHandleState(pipe, Some(&mode), None, None) {
            let _ = CloseHandle(pipe);
            return Err(PipeError::Win32((e.code().0 as u32) & 0xFFFF));
        }
    }
    if let Err(e) = write_frame(pipe, payload) {
        let _ = unsafe { CloseHandle(pipe) };
        return Err(e);
    }
    let _ = unsafe { FlushFileBuffers(pipe) };
    let _ = unsafe { CloseHandle(pipe) };
    Ok(())
}
```

### `dlp-agent/src/bypass_correlator.rs` (service, event-driven)

**Analog:** `dlp-agent/src/bypass_correlator.rs` (existing file)

**Existing run() task spawn pattern** (lines 991-1049):
```rust
pub async fn run(
    self,
    etw_rx: Receiver<EtwFileEvent>,
    process_rx: Receiver<ProcessEvent>,
    bypass_rx: Receiver<BypassAlert>,
    server_client: ServerClient,
) {
    let correlator = Arc::new(self);

    // Task 1: Process event handler.
    let proc_corr = Arc::clone(&correlator);
    let proc_handle = tokio::spawn(async move {
        while let Ok(event) = process_rx.recv() {
            proc_corr.handle_process_event(event).await;
        }
    });

    // Task 2: ETW event handler.
    let etw_corr = Arc::clone(&correlator);
    let etw_handle = tokio::spawn(async move {
        while let Ok(event) = etw_rx.recv() {
            etw_corr.handle_etw_event(event).await;
        }
    });

    // Task 3: Bypass alert handler (from hook DLL IPC).
    let bypass_corr = Arc::clone(&correlator);
    let bypass_handle = tokio::task::spawn_blocking(move || {
        while let Ok(alert) = bypass_rx.recv() {
            let bypass_corr = Arc::clone(&bypass_corr);
            if let Ok(rt) = tokio::runtime::Handle::try_current() {
                rt.block_on(bypass_corr.submit_bypass_alert(alert));
            }
        }
    });

    // Task 4: Batch flush task.
    let flush_corr = Arc::clone(&correlator);
    let flush_handle = tokio::spawn(async move {
        let mut interval = tokio::time::interval(Duration::from_secs(flush_corr.config.flush_interval_secs));
        loop {
            interval.tick().await;
            flush_corr.flush_batch(&server_client).await;
        }
    });

    tokio::select! {
        _ = proc_handle => {},
        _ = etw_handle => {},
        _ = bypass_handle => {},
        _ = flush_handle => {},
    }
}
```

```rust
pub fn insert_batch(
    uow: &UnitOfWork<'_>,
    rows: &[BypassAlertInsertRow],
) -> rusqlite::Result<(usize, usize)> {
    let mut inserted: usize = 0;
    let mut skipped: usize = 0;
    for row in rows {
        let last_id = Self::insert(uow, row)?;
        if last_id == 0 {
            skipped += 1;
        } else {
            inserted += 1;
        }
    }
    Ok((inserted, skipped))
}
```
