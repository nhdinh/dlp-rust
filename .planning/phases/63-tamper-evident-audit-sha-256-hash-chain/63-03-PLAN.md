  <action>
    1. **Sort events within the batch by (agent_id, event_timestamp)** before verification to prevent false positives from out-of-order arrival:
       ```rust
       // Sort by agent_id, then timestamp to ensure correct per-agent chain order
       events.sort_by(|a, b| {
           a.agent_id.cmp(&b.agent_id)
               .then_with(|| a.event_timestamp.cmp(&b.event_timestamp))
       });
       ```
       **Out-of-order policy:** Events that arrive out of order within a batch are reordered before verification. Events from different agents are independent and never cause chain breaks. If an event's `prev_hash` does not match the expected value (after sorting), it is flagged as a chain break. Gap-fill detection is deferred to a future phase.

    2. After the app-identity validation loop and before `let count = events.len();`, add a chain verification pass. Clone the pool early so it is available:
       ```rust
       let pool = Arc::clone(&state.pool);

       let mut chain_breaks = Vec::new();
       let mut last_hash_cache: std::collections::HashMap<String, String> = std::collections::HashMap::new();

       for event in &events {
           if let Some(ref chain_hash) = event.chain_hash {
               let prev_hash = event.prev_hash.as_deref().unwrap_or_else(|| dlp_common::audit::genesis_hash());

               // Step A: Verify event integrity (recomputed hash must match claimed chain_hash)
               match dlp_common::audit::compute_chain_hash(prev_hash, event) {
                   Ok(expected) => {
                       if expected != *chain_hash {
                           chain_breaks.push((
                               event.agent_id.clone(),
                               event.correlation_id.clone(),
                               ChainBreakReason::HashMismatch,
                           ));
                           continue;
                       }
                   }
                   Err(e) => {
                       tracing::error!(error = %e, agent_id = %event.agent_id, "failed to compute chain hash; rejecting event");
                       chain_breaks.push((
                           event.agent_id.clone(),
                           event.correlation_id.clone(),
                           ChainBreakReason::HashComputationFailed,
                       ));
                       continue;
                   }
               }

               // Step B: Verify chain continuity (prev_hash must match last stored hash for this agent)
               // Use in-memory cache first, then fall back to DB query
               let expected_prev = last_hash_cache.get(&event.agent_id).cloned()
                   .or_else(|| {
                       AuditEventRepository::get_last_chain_hash(&pool, &event.agent_id).ok().flatten()
                   })
                   .unwrap_or_else(dlp_common::audit::genesis_hash);

               if prev_hash != expected_prev {
                   chain_breaks.push((
                       event.agent_id.clone(),
                       event.correlation_id.clone(),
                       ChainBreakReason::PrevHashMismatch,
                   ));
               } else {
                   // Update cache with this event's chain_hash for subsequent events in the same batch
                   last_hash_cache.insert(event.agent_id.clone(), chain_hash.clone());
               }
           }
       }
       ```
       **Two-step verification clarification:**
       - Step A (Integrity): The server recomputes `compute_chain_hash(event.prev_hash, event)` and compares it to the event's *claimed* `chain_hash`. This detects payload tampering.
       - Step B (Continuity): The server compares `event.prev_hash` to the last stored `chain_hash` for that `agent_id` (from DB or in-batch cache). This detects deletion or reordering.

       Define the reason enum:
       ```rust
       #[derive(Debug, Clone, PartialEq)]
       enum ChainBreakReason {
           HashMismatch,           // Recomputed hash != claimed chain_hash
           PrevHashMismatch,       // event.prev_hash != last stored chain_hash
           HashComputationFailed,  // compute_chain_hash returned Err
       }
       ```

    4. Pass `chain_breaks` into the spawn_blocking closure by cloning it:

       ```rust
       let chain_breaks_for_block = chain_breaks.clone();
       ```

    5. After the spawn_blocking returns and before SIEM relay, trigger tamper alerts AND persist synthetic events:
       ```rust
       if !chain_breaks.is_empty() {
           let alert_router = Arc::clone(&state.alert);
           let pool_for_alert = Arc::clone(&state.pool);

           tokio::spawn(async move {
               // Deduplicate: one alert per (agent_id, reason) per batch
               let mut seen = std::collections::HashSet::new();
               for (agent_id, correlation_id, reason) in &chain_breaks {
                   let key = (agent_id.clone(), format!("{:?}", reason));
                   if !seen.insert(key) {
                       continue; // skip duplicate
                   }

                   let synthetic = dlp_common::AuditEvent::new(
                       dlp_common::EventType::ChainBreakDetected,
                       "S-1-5-18".to_string(),
                       "SYSTEM".to_string(),
                       "audit_chain_break".to_string(),
                       dlp_common::Classification::T4,
                       dlp_common::Action::WRITE,
                       dlp_common::Decision::DenyWithAlert,
                       agent_id.clone(),
                       0,
                   )
                   .with_correlation_id(correlation_id.clone());

                   // Persist the synthetic event to the audit log (completes the audit trail)
                   if let Err(e) = tokio::task::spawn_blocking({
                       let pool = Arc::clone(&pool_for_alert);
                       let event = synthetic.clone();
                       move || {
                           let mut conn = pool.get()?;
                           let row = AuditEventRow { /* ... map synthetic event fields ... */ };
                           AuditEventRepository::insert_batch(&mut conn, &[row])
                       }
                   }).await {
                       tracing::error!(error = %e, agent_id = %agent_id, "failed to persist chain break event to audit log");
                   }

                   if let Err(e) = alert_router.send_alert(&synthetic).await {
                       tracing::error!(error = %e, agent_id = %agent_id, "failed to send chain break alert");
                   }

                   tracing::error!(
                       agent_id = %agent_id,
                       correlation_id = %correlation_id.as_deref().unwrap_or("none"),
                       reason = ?reason,
                       "audit chain break detected"
                   );
               }
           });
       }
       ```
       **Synthetic event persistence:** The synthetic `ChainBreakDetected` event is inserted into `audit_events` with `Decision::DenyWithAlert`. This preserves the detection in the tamper-evident log itself, fulfilling the compliance requirement that the audit trail must record when breaks were detected.
