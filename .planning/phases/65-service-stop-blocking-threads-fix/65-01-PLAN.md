Watchdog design:
```rust
// Spawned before joining begins. Sleeps for SHUTDOWN_TIMEOUT + 5s.
// If main thread is still alive, calls std::process::exit(1).
std::thread::spawn(move || {
    std::thread::sleep(SHUTDOWN_TIMEOUT + Duration::from_secs(5));
    // If we're still running, something is stuck. Force exit.
    error!("shutdown watchdog: forced process exit after timeout");
    std::process::exit(1);
});
```

Join strategy:
```rust
fn shutdown_and_join(self, timeout_per_thread: Duration) {
    request_shutdown();
    info!("shutdown requested — joining blocking threads");

    let start = Instant::now();

    let join_with_timeout = |name: &str, handle: Option<JoinHandle<()>>| {
        if let Some(h) = handle {
            debug!(thread = name, "joining thread");
            match h.join() {
                Ok(()) => {
                    let elapsed = start.elapsed();
                    debug!(thread = name, ?elapsed, "thread joined cleanly");
                }
                Err(e) => {
                    warn!(thread = name, error = ?e, "thread panicked during shutdown");
                }
            }
        }
    };

    join_with_timeout("health", self.health);
    for (i, h) in self.ipc.into_iter().enumerate() {
        join_with_timeout(&format!("ipc-pipe-{i}"), Some(h));
    }
    join_with_timeout("chrome", self.chrome);
    join_with_timeout("session", self.session);

    let total = start.elapsed();
    info!(?total, "all blocking threads joined");
}
```

Note: We do NOT implement a per-thread join timeout because std::thread::JoinHandle::join()
has no native timeout. The watchdog provides the safety net. If a specific thread is known
to be stuck, the watchdog ensures the process exits. Future improvement: CancelSynchronousIo
or overlapped I/O for per-thread cancellation.

    /// All blocking threads must poll this flag and break their loops.
    ///
    /// Reset to false at service startup to support in-process restart scenarios.
    static SHUTDOWN_REQUESTED: AtomicBool = AtomicBool::new(false);

    /// Returns true if service shutdown has been authorized.
    pub fn shutdown_requested() -> bool {
        SHUTDOWN_REQUESTED.load(Ordering::Acquire)
    }

    /// Requests service shutdown. Idempotent — safe to call multiple times.
    pub fn request_shutdown() {
        SHUTDOWN_REQUESTED.store(true, Ordering::Release);
    }

    /// Resets the shutdown signal to false. Called at service startup.
    pub fn reset_shutdown_signal() {
        SHUTDOWN_REQUESTED.store(false, Ordering::Release);
    }

    /// Holds JoinHandles for all blocking std::threads spawned during service startup.
    /// Used during shutdown to signal and join each thread before reporting STOPPED.
    struct BlockingThreads {
        health: Option<std::thread::JoinHandle<()>>,
        ipc: Vec<std::thread::JoinHandle<()>>,
        chrome: Option<std::thread::JoinHandle<()>>,
        session: Option<std::thread::JoinHandle<()>>,
    }

    impl BlockingThreads {
        fn new() -> Self {
            Self {
                health: None,
                ipc: Vec::new(),
                chrome: None,
                session: None,
            }
        }

        /// Signal shutdown and join all threads. A watchdog timer guarantees
        /// the process exits if any thread remains blocked beyond the timeout.
        fn shutdown_and_join(self, timeout_per_thread: Duration) {
            request_shutdown();
            info!("shutdown requested — joining blocking threads");

            // Watchdog: if shutdown takes longer than timeout + 5s buffer, force exit.
            // This prevents the service from hanging in StopPending indefinitely
            // when a thread is blocked in a Win32 call that never returns.
            let watchdog_timeout = timeout_per_thread.saturating_mul(4)
                .saturating_add(Duration::from_secs(5));
            std::thread::spawn(move || {
                std::thread::sleep(watchdog_timeout);
                error!("shutdown watchdog: forced process exit after {:?}", watchdog_timeout);
                std::process::exit(1);
            });

            let start = std::time::Instant::now();

            let join_with_log = |name: &str, handle: Option<std::thread::JoinHandle<()>>| {
                if let Some(h) = handle {
                    debug!(thread = name, "joining thread");
                    match h.join() {
                        Ok(()) => {
                            let elapsed = start.elapsed();
                            debug!(thread = name, ?elapsed, "thread joined cleanly");
                        }
                        Err(e) => {
                            warn!(thread = name, error = ?e, "thread panicked during shutdown");
                        }
                    }
                }
            };

            join_with_log("health", self.health);
            for (i, h) in self.ipc.into_iter().enumerate() {
                join_with_log(&format!("ipc-pipe-{i}"), Some(h));
            }
            join_with_log("chrome", self.chrome);
            join_with_log("session", self.session);

            let total = start.elapsed();
            info!(?total, "all blocking threads joined");
        }
    }
