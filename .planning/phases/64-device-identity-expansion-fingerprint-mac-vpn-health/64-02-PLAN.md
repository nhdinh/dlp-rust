       ```rust
       //! Device identity collection via Windows APIs.
       //!
       //! Provides functions to collect:
       //! - MAC addresses (GetAdaptersAddresses)
       //! - VPN state (adapter enumeration + heuristics)
       //! - Domain join state (NetGetJoinInformation)
       //! - Device fingerprint (SHA-256 of stable hardware attributes)

       use dlp_common::endpoint::{DeviceFingerprint, EndpointIdentity, MacAddress};
       use sha2::{Digest, Sha256};
       use tracing::{error, info, warn};

       /// Errors that can occur during device info collection.
       #[derive(Debug, thiserror::Error)]
       pub enum DeviceInfoError {
           #[error("Win32 API error: {0}")]
           Win32(#[from] windows::core::Error),
           #[error("network adapter enumeration failed: {0}")]
           AdapterEnumeration(String),
           #[error("domain join detection failed: {0}")]
           DomainDetection(String),
           #[error("fingerprint computation failed: {0}")]
           FingerprintComputation(String),
       }

       /// Collects MAC addresses of all active physical network interfaces.
       #[cfg(windows)]
       pub fn collect_mac_addresses() -> Result<Vec<MacAddress>, DeviceInfoError> {
           // Use GetAdaptersAddresses(AF_UNSPEC, GAA_FLAG_INCLUDE_PREFIX, ...)
           // Filter out:
           // - Loopback interfaces (IF_TYPE_SOFTWARE_LOOPBACK)
           // - Virtual/Tunnel interfaces (IF_TYPE_TUNNEL, IF_TYPE_PROP_VIRTUAL)
           // - Interfaces with OperStatus != IfOperStatusUp
           // Return MACs as colon-separated hex strings
       }

       /// Detects whether the endpoint is connected via VPN.
       #[cfg(windows)]
       pub fn detect_vpn_state() -> Result<bool, DeviceInfoError> {
           // Heuristic: look for adapters with:
           // - IfType == IF_TYPE_TUNNEL (131)
           // - Description containing "VPN", "TAP", "WireGuard", "OpenVPN", "Cisco AnyConnect"
           // - OR use existing get_network_location from dlp-common as fallback
       }

       /// Detects domain join state and returns (is_joined, domain_name).
       #[cfg(windows)]
       pub fn detect_domain_join() -> Result<(bool, String), DeviceInfoError> {
           // Use NetGetJoinInformation(NULL, &name_buf, &buffer_type)
           // NetSetupDomainName = joined, NetSetupWorkgroupName = not joined
           // Free name with NetApiBufferFree
       }

       /// Computes the device fingerprint from stable hardware attributes.
       #[cfg(windows)]
       pub fn compute_fingerprint(
           hostname: &str,
           macs: &[MacAddress],
           os_version: &str,
       ) -> Result<DeviceFingerprint, DeviceInfoError> {
           // Read InstallDate from registry:
           // HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\InstallDate (DWORD, Unix timestamp)
           // Fallback: use agent install path metadata (std::fs::metadata on agent exe parent dir)
           // SHA-256(hostname + "|" + sorted_macs_joined + "|" + os_version + "|" + install_date)
       }

       /// Builds a complete EndpointIdentity by calling all collection functions.
       #[cfg(windows)]
       pub fn collect_endpoint_identity() -> Result<EndpointIdentity, DeviceInfoError> {
           let hostname = std::env::var("COMPUTERNAME").unwrap_or_else(|_| "unknown".into());
           let os_version = get_os_version()?;
           let macs = collect_mac_addresses()?;
           let fingerprint = compute_fingerprint(&hostname, &macs, &os_version)?;
           let (domain_joined, domain_name) = detect_domain_join()?;
           let vpn_connected = detect_vpn_state()?;

           Ok(EndpointIdentity {
               fingerprint,
               mac_addresses: macs,
               vpn_connected,
               domain_joined,
               domain_name,
               health_status: dlp_common::endpoint::DeviceHealthStatus::Healthy,
               hostname,
               os_version,
           })
       }

       /// Returns the OS version string.
       #[cfg(windows)]
       fn get_os_version() -> Result<String, DeviceInfoError> {
           // Use RtlGetVersion or read from registry
           // HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\DisplayVersion
           // or ProductName + ReleaseId
       }
       ```

    ```rust
    use windows::Win32::Networking::WinSock::{AF_UNSPEC, SOCKET_ADDRESS};
    use windows::Win32::NetworkManagement::IpHelper::{
        GetAdaptersAddresses, GAA_FLAG_INCLUDE_PREFIX, IP_ADAPTER_ADDRESSES_LH,
    };
    use windows::Win32::NetworkManagement::Ndis::{
        IF_TYPE_PROP_VIRTUAL, IF_TYPE_SOFTWARE_LOOPBACK, IF_TYPE_TUNNEL,
    };

    // SAFETY: GetAdaptersAddresses with null buffer first call gets required size.
    // Allocate buffer, call again to fill.
    // Walk linked list via .Next() pointer.
    // Filter: PhysicalAddressLength == 6 (Ethernet MAC), not loopback, not tunnel, not virtual.
    // Format as colon-separated hex: AA:BB:CC:DD:EE:FF
    ```

    - Skip IfType == IF_TYPE_SOFTWARE_LOOPBACK (24)

    - Skip IfType == IF_TYPE_PROP_VIRTUAL (53)
    - Skip OperStatus != IfOperStatusUp (1)
    - Only include adapters with PhysicalAddressLength == 6

    Log a warning if no MACs are found (unexpected but not fatal).
