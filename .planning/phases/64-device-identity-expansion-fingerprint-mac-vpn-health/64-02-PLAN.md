  <action>
    Add the following functions to dlp-agent/src/device_identity.rs:

    1. compute_fingerprint(hostname: &str, mac_addresses: &[String], os_version: &str, install_date: &str) -> String
       - Sort mac_addresses lexicographically (clone then sort)
       - Join sorted MACs with ","
       - Concatenate with v1 prefix: format!("v1:{}|{}|{}|{}", hostname, macs_joined, os_version, install_date)
       - Compute SHA-256 using sha2::Sha256 + sha2::Digest
       - Return lowercase hex: format!("{:x}", hasher.finalize())
       - This is a pure function — no Windows API calls, no cfg gating needed
       - The "v1:" prefix addresses review concern about fingerprint versioning

    2. read_fingerprint_from_registry() -> Option<String>
       - Windows: Open HKLM\SOFTWARE\DLP\Agent with KEY_READ
       - Query "device_fingerprint" value (REG_SZ)
       - Return Some(value) on success, None on any error
       - Non-Windows: return None

    3. write_fingerprint_to_registry(fingerprint: &str) -> anyhow::Result<()>
       - Windows: Use RegCreateKeyExW to create/open HKLM\SOFTWARE\DLP\Agent
       - Use RegSetValueExW to write "device_fingerprint" as REG_SZ
       - Convert fingerprint to UTF-16 null-terminated wide string: fingerprint.encode_utf16().chain(std::iter::once(0)).collect()
       - Close key with RegCloseKey
       - Return Ok(()) on success, Err(anyhow::anyhow!(...)) on failure
       - Non-Windows: return Ok(()) (no-op)

    4. read_install_date_from_registry() -> Option<String>
       - Windows: Read HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\InstallDate (DWORD, Unix epoch seconds)
       - Convert to ISO-8601 date string using chrono (e.g., "2024-01-15")
       - Non-Windows: return None

    5. get_os_version_string() -> String
       - Windows: Read HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion, concatenate "ProductName" + " " + "DisplayVersion"
       - Fallback to std::env::var("OS").unwrap_or_else(|_| "Windows".to_string()) if registry fails
       - Non-Windows: return "non-windows (test)".to_string()

    Add the required imports:
    - use sha2::{Digest, Sha256};
    - use windows::Win32::System::Registry::{RegCloseKey, RegCreateKeyExW, RegOpenKeyExW, RegQueryValueExW, RegSetValueExW, HKEY_LOCAL_MACHINE, KEY_READ, KEY_WRITE, REG_SZ};
    - use windows::Win32::NetworkManagement::NetManagement::{NetApiBufferFree, NetGetJoinInformation, NETSETUP_JOIN_STATUS};
    - use windows::Win32::NetworkManagement::IpHelper::{GetAdaptersAddresses, GAA_FLAG_INCLUDE_PREFIX, IP_ADAPTER_ADDRESSES_LH};
    - use windows::Win32::Networking::WinSock::AF_UNSPEC;
    - use windows::Win32::NetworkManagement::Ndis::{IF_TYPE_TUNNEL};
