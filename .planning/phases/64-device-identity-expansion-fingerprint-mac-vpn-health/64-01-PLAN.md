    /// Computed from hostname + MAC addresses + OS version + install date
    /// so that reinstalling the agent on the same hardware produces the
    /// same fingerprint.
    #[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, Default)]

        /// Returns the fingerprint as a string slice.
        #[must_use]

    ```rust
    /// A MAC address in colon-separated hex format (e.g. "AA:BB:CC:DD:EE:FF").
    #[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, Default)]
    #[serde(transparent)]
    pub struct MacAddress(pub String);

    impl MacAddress {
        /// Returns the MAC address as a string slice.
        #[must_use]
        pub fn as_str(&self) -> &str {
            &self.0
        }

        /// Validates that the string is a well-formed MAC address.
        /// Accepts colon-separated, hyphen-separated, or no-separator formats.
        #[must_use]
        pub fn is_valid(&self) -> bool {
            let s = self.0.replace([':', '-'], "");
            s.len() == 12 && s.chars().all(|c| c.is_ascii_hexdigit())
        }
    }

    impl From<String> for MacAddress {
        fn from(s: String) -> Self {
            Self(s)
        }
    }
    ```
