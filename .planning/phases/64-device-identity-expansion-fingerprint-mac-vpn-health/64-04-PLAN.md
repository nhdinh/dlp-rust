       /// Atomically transitions the device health status.
       ///
       /// Returns the previous health status. Emits an audit event if the status changed.
       /// All tamper/connectivity detection paths MUST call this function to avoid races.
       ///
       /// # Async Safety
       /// This function is sync. When called from async code (the heartbeat loop), the caller
       /// MUST wrap the registry write in `tokio::task::spawn_blocking` to avoid blocking the
       /// Tokio reactor. The `transition_health_async` wrapper handles this.
       ///
       /// # Eventual Consistency Note
       /// The health status read by `current_health()` is a point-in-time snapshot. Between
       /// the read and the heartbeat send, the status may change. This is intentional -- the
       /// stale-read window is minimized by calling `current_health()` immediately before
       /// serialization, but true snapshot consistency across the full send path is not
       /// guaranteed. Consumers MUST treat health status as eventually consistent.
       pub fn transition_health(new_status: DeviceHealthStatus) -> DeviceHealthStatus {
           let new_u8 = health_to_u8(new_status);
           let prev_u8 = HEALTH_STATUS.swap(new_u8, Ordering::SeqCst);
           let prev_status = u8_to_health(prev_u8);
           if prev_u8 != new_u8 {
               tracing::info!(prev = ?prev_status, new = ?new_status, "device health status changed");
               // Persist to registry so restart preserves state (addresses review concern)
               // Note: this is a sync registry write. Callers from async MUST use
               // transition_health_async() instead.
               let _ = write_health_status_to_registry(&new_status);
           }
           prev_status
       }
       
       /// Async-safe wrapper for transition_health.
       ///
       /// Wraps the registry write in spawn_blocking to avoid blocking the async runtime.
       /// Use this when calling from the heartbeat loop or other async contexts.
       pub async fn transition_health_async(new_status: DeviceHealthStatus) -> DeviceHealthStatus {
           let prev = transition_health(new_status);
           // The registry write inside transition_health is sync. If we are in an async context,
           // re-do the registry write in spawn_blocking to be safe.
           let status = new_status;
           let _ = tokio::task::spawn_blocking(move || {
               write_health_status_to_registry(&status)
           }).await;
           prev
       }
       
       /// Returns the current device health status.
       pub fn current_health() -> DeviceHealthStatus {
           u8_to_health(HEALTH_STATUS.load(Ordering::SeqCst))
       }
       
       /// Reads health status from registry at startup.
       /// Call this in service.rs init to restore state after restart.
       pub fn read_health_from_registry() -> Option<DeviceHealthStatus> { ... }
       
       /// Writes health status to registry.
       fn write_health_status_to_registry(status: &DeviceHealthStatus) -> anyhow::Result<()> { ... }

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Agent health state | AtomicU8 prevents race conditions across tamper/connectivity detection paths |
| ABAC evaluation | DeviceHealth condition is one signal among many; never sole decision factor |
| Audit log | Health transitions are append-only audit events with prev/new state |
| Registry persistence | Health status survives agent restart |
| Async safety | spawn_blocking prevents registry I/O from blocking the Tokio reactor |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-64-11 | Tampering | Health status manipulated by malware | accept | Health status is telemetry, not a security boundary; ABAC uses it as one condition among many |
| T-64-12 | Denial of Service | Rapid health transitions flood audit log | mitigate | Atomic swap prevents spurious transitions; only actual changes emit events |
| T-64-13 | Elevation of Privilege | Agent reports Healthy after tamper | mitigate | Tamper detection paths call report_tamper_detected() which sets Tampered; recovery to Healthy requires successful heartbeat |
| T-64-14 | Repudiation | Health transition not audited | mitigate | Every transition emits AuditEvent with EventType::DeviceHealthChange |
| T-64-15 | Information Disclosure | Health status in registry | accept | Health status is not sensitive; HKLM access requires admin |
| T-64-16 | Denial of Service | Registry write blocks heartbeat loop | mitigate | Async wrapper uses spawn_blocking; sync path is only for non-async callers |
</threat_model>
