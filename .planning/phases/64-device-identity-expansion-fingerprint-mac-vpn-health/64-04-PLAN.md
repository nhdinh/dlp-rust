         ```rust
                 DeviceHealthStatus::Tampered => 3,
             }
         }
         fn u8_to_health(v: u8) -> DeviceHealthStatus {
             match v {

                 3 => DeviceHealthStatus::Tampered,
                 _ => DeviceHealthStatus::Healthy,
             }
         }
         ```

    2. Add a health transition function to dlp-agent/src/device_identity.rs:
       ```rust
       /// Atomically transitions the device health status.
       ///
       /// Returns the previous health status. Emits an audit event if the status changed.
       /// All tamper/connectivity detection paths MUST call this function to avoid races.
       pub fn transition_health(new_status: DeviceHealthStatus) -> DeviceHealthStatus {
           let new_u8 = health_to_u8(new_status);
           let prev_u8 = HEALTH_STATUS.swap(new_u8, Ordering::SeqCst);
           let prev_status = u8_to_health(prev_u8);
           if prev_u8 != new_u8 {
               tracing::info!(prev = ?prev_status, new = ?new_status, "device health status changed");
               // Audit event emission is handled by the caller or a background task
           }
           prev_status
       }
       
       /// Returns the current device health status.
       pub fn current_health() -> DeviceHealthStatus {
           u8_to_health(HEALTH_STATUS.load(Ordering::SeqCst))
       }
       ```
       - Move health_to_u8 and u8_to_health to device_identity.rs as well (they are used by transition_health)
       - HEALTH_STATUS static stays in service.rs; transition_health and current_health are in device_identity.rs and reference the static via a pub(crate) re-export or pass it as a parameter. Simpler: move HEALTH_STATUS to device_identity.rs and make transition_health/current_health pub.
