## Decision constraints
- D-03: If PnP disable fails, log ERROR and return Err. Only Ok(()) for CR_NO_SUCH_DEVNODE after successful initial locate.
- D-06: If PnP disable fails, DACL defense-in-depth MUST still attempt. If BOTH fail, return unified Err.
- D-07: All enforcement outcomes emit structured tracing span with fields: vid, pid, serial, drive_letter, tier, pnp_result, dacl_result.
- D-08: Scope is PnP disable fix ONLY. No disk/ABAC changes.
- D-09: Fix applies to both hot-plug arrivals and startup scan.

    ```rust
    #[cfg(windows)]
    fn apply_tier_enforcement(
        letter: char,
        identity: &DeviceIdentity,
        dbcc_name: &str,
    ) -> Result<(), String> {
        let Some(controller) = DEVICE_CONTROLLER.get() else {
            return Err("DeviceController not initialized".to_string());
        };

        let tier = if let Some(cache) = REGISTRY_CACHE.get() {
            cache.trust_tier_for(&identity.vid, &identity.pid, &identity.serial)
        } else {
            warn!("registry cache unavailable — applying default-deny (Blocked)");
            UsbTrustTier::Blocked
        };

        match tier {
            UsbTrustTier::Blocked => {
                // Layer 1: PnP disable (primary enforcement).
                let pnp_result = controller.disable_usb_device(dbcc_name, identity);
                let pnp_ok = pnp_result.is_ok();
                if let Err(ref e) = pnp_result {
                    error!(
                        vid = %identity.vid,
                        pid = %identity.pid,
                        serial = %identity.serial,
                        drive = %letter,
                        tier = "Blocked",
                        pnp_result = "err",
                        error = %e,
                        "PnP disable failed for blocked USB device"
                    );
                }

                // Layer 2: DACL deny-all (defense-in-depth fallback — always attempt).
                let dacl_result = controller.set_volume_deny_all(letter);
                let dacl_ok = dacl_result.is_ok();
                if let Err(ref e) = dacl_result {
                    error!(
                        drive = %letter,
                        dacl_result = "err",
                        error = %e,
                        "DACL deny-all failed for blocked USB device"
                    );
                }

                // Structured outcome span (D-07).
                if pnp_ok && dacl_ok {
                    info!(
                        vid = %identity.vid,
                        pid = %identity.pid,
                        serial = %identity.serial,
                        drive = %letter,
                        tier = "Blocked",
                        pnp_result = "ok",
                        dacl_result = "ok",
                        "USB device blocked (PnP + DACL)"
                    );
                    Ok(())
                } else if !pnp_ok && !dacl_ok {
                    error!(
                        vid = %identity.vid,
                        pid = %identity.pid,
                        serial = %identity.serial,
                        drive = %letter,
                        tier = "Blocked",
                        pnp_result = "err",
                        dacl_result = "err",
                        "USB device block FAILED — both PnP disable and DACL deny-all failed"
                    );
                    Err(format!(
                        "Both PnP disable ({}) and DACL deny-all ({}) failed",
                        pnp_result.unwrap_err(),
                        dacl_result.unwrap_err()
                    ))
                } else {
                    // One succeeded, one failed — log warning but return Ok (defense-in-depth worked).
                    warn!(
                        vid = %identity.vid,
                        pid = %identity.pid,
                        serial = %identity.serial,
                        drive = %letter,
                        tier = "Blocked",
                        pnp_result = if pnp_ok { "ok" } else { "err" },
                        dacl_result = if dacl_ok { "ok" } else { "err" },
                        "USB device block partially succeeded (defense-in-depth)"
                    );
                    Ok(())
                }
            }
            UsbTrustTier::ReadOnly => {
                let result = controller.set_volume_readonly(letter);
                match result {
                    Ok(()) => {
                        info!(
                            vid = %identity.vid,
                            pid = %identity.pid,
                            serial = %identity.serial,
                            drive = %letter,
                            tier = "ReadOnly",
                            "USB volume set to read-only"
                        );
                        Ok(())
                    }
                    Err(e) => {
                        error!(
                            vid = %identity.vid,
                            pid = %identity.pid,
                            serial = %identity.serial,
                            drive = %letter,
                            tier = "ReadOnly",
                            error = %e,
                            "Failed to set USB volume to read-only"
                        );
                        Err(format!("ReadOnly DACL failed: {e}"))
                    }
                }
            }
            UsbTrustTier::FullAccess => {
                debug!(
                    vid = %identity.vid,
                    pid = %identity.pid,
                    serial = %identity.serial,
                    drive = %letter,
                    tier = "FullAccess",
                    "USB device has FullAccess — no enforcement action"
                );
                Ok(())
            }
        }
    }
    ```

    Important: The old `apply_tier_enforcement` returned `()`. The new one returns `Result<(), String>`. All call sites must be updated to handle the Result.
