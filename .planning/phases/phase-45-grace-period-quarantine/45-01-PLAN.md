<objective>
Implement configurable grace period for unregistered disk arrivals. When a disk arrives and `disk_grace_period_seconds > 0`, mount the disk normally but block writes. After the grace period expires, apply mount-time blocking (drive letter removal + volume offline). Emit audit events and toast notifications.

Purpose: Give users a configurable window to register new disks before hard enforcement kicks in.
Output: Grace period logic integrated into disk arrival handling with config-driven timeout.
</objective>

3. Add `start_grace_period` function:
```rust
fn start_grace_period(
    letter: char,
    disk: &DiskIdentity,
    grace_seconds: u64,
    audit_ctx: &EmitContext,
) {
    if grace_seconds == 0 {
        // Immediate block (Phase 44 behavior)
        let _ = block_disk_at_mount_time(letter, disk, audit_ctx);
        return;
    }

    // Insert into drive_letter_map so disk is accessible
    let enumerator = get_disk_enumerator();
    {
        let mut map = enumerator.drive_letter_map.write();
        map.insert(letter, disk.clone());
    }

    // Track grace period
    {
        let mut gp_map = enumerator.grace_period_map.write();
        gp_map.insert(disk.instance_id.clone(), Instant::now());
    }

    // Emit audit event
    emit_disk_quarantine_started(disk, grace_seconds, audit_ctx);

    // Spawn timer to enforce block after grace period
    let instance_id = disk.instance_id.clone();
    let audit_ctx_clone = audit_ctx.clone();
    tokio::spawn(async move {
        tokio::time::sleep(Duration::from_secs(grace_seconds)).await;
        enforce_grace_period_expiry(&instance_id, &audit_ctx_clone);
    });
}
```

4. Add `enforce_grace_period_expiry`:
```rust
fn enforce_grace_period_expiry(instance_id: &str, audit_ctx: &EmitContext) {
    let enumerator = get_disk_enumerator();

    // Check if disk is still connected
    let disk = {
        let map = enumerator.drive_letter_map.read();
        map.values().find(|d| d.instance_id == instance_id).cloned()
    };

    let Some(disk) = disk else { return };

    // Remove from grace_period_map
    {
        let mut gp_map = enumerator.grace_period_map.write();
        gp_map.remove(instance_id);
    }

    // Apply mount-time block
    if let Some(letter) = disk.drive_letter {
        let _ = block_disk_at_mount_time(letter, &disk, audit_ctx);
    }

    // Remove from drive_letter_map
    {
        let mut map = enumerator.drive_letter_map.write();
        if let Some(letter) = disk.drive_letter {
            map.remove(&letter);
        }
    }

    emit_disk_quarantine_expired(&disk, audit_ctx);
}
```

```rust
fn on_disk_arrival_inner(device_path: &str, live_disks: &[DiskIdentity], audit_ctx: &EmitContext) {
    let enumerator = get_disk_enumerator();
    let instance_id_map = enumerator.instance_id_map.read();
    let config = /* get config */; // Need config access
    let grace_seconds = config.disk_grace_period_seconds;

    for disk in live_disks {
        let Some(letter) = disk.drive_letter else { continue };

        if !instance_id_map.contains_key(&disk.instance_id) {
            // Unregistered disk — start grace period
            start_grace_period(letter, disk, grace_seconds, audit_ctx);
            continue;
        }

        // Registered disk — normal flow
        {
            let mut map = enumerator.drive_letter_map.write();
            if map.contains_key(&letter) { continue; }
            map.insert(letter, disk.clone());
        }
    }
}
```
