2. In `dlp-agent/src/service.rs`, add a periodic task loop inside `run_loop` (or as a separate tokio task spawned from run_loop):

   ```rust
   let registry_for_cleanup = Arc::clone(&process_registry);
   let registry_for_telemetry = Arc::clone(&process_registry);
   let injector_for_backstop = injector.clone(); // if available
   let matcher_for_backstop = allowlist_matcher.clone(); // if available

   tokio::spawn(async move {
       let mut cleanup_interval = tokio::time::interval(Duration::from_secs(60));
       let mut backstop_interval = tokio::time::interval(Duration::from_secs(300));
       let mut telemetry_interval = tokio::time::interval(Duration::from_secs(60));

       loop {
           tokio::select! {
               _ = cleanup_interval.tick() => {
                   let removed = registry_for_cleanup.cleanup_exited();
                   if removed > 0 {
                       debug!(removed, "periodic cleanup sweep removed exited PIDs");
                   }
               }
               _ = backstop_interval.tick() => {
                   debug!("periodic EnumProcesses backstop sweep starting");
                   if let Err(e) = crate::universal_injector::startup_enum_processes_sweep(
                       &injector_for_backstop,
                       &registry_for_cleanup,
                       &matcher_for_backstop,
                   ) {
                       warn!(error = %e, "periodic backstop sweep encountered errors");
                   }
               }
               _ = telemetry_interval.tick() => {
                   let snapshot = registry_for_telemetry.telemetry_snapshot();
                   info!(
                       injected = snapshot.injected_count,
                       skipped = snapshot.total_tracked - snapshot.injected_count,
                       coverage_percent = snapshot.coverage_percent,
                       "injection telemetry"
                   );
                   // Emit siem.injection_telemetry audit event
                   if let Some(ref siem_tx) = siem_tx {
                       let event = dlp_common::AuditEvent {
                           event_type: "injection_telemetry".to_string(),
                           user_sid: "".to_string(),
                           user_name: "".to_string(),
                           resource_path: format!("injected={},skipped={},coverage={:.1}%", 
                               snapshot.injected_count, 
                               snapshot.total_tracked - snapshot.injected_count,
                               snapshot.coverage_percent),
                           classification: "".to_string(),
                           action_attempted: "telemetry".to_string(),
                           decision: "ok".to_string(),
                           policy_id: None,
                           policy_name: None,
                           agent_id: hostname.clone().unwrap_or_default(),
                           session_id: 0,
                           access_context: "local".to_string(),
                           correlation_id: None,
                       };
                       let _ = siem_tx.send(event).await;
                   }
               }
           }
       }
   });
   ```

   NOTE: The `siem_tx` and `hostname` variables must be accessible in this scope. If they are not, adjust the closure captures accordingly.

      ```powershell
      # AppInit_DLLs tertiary fallback setup (BLOCK-07)
      $appInitKey = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Windows"
      $backupKey = "HKLM:\SOFTWARE\DLP\Backup\AppInit_DLLs"

      # Create backup key if not exists
      if (-not (Test-Path $backupKey)) {
          New-Item -Path $backupKey -Force | Out-Null
      }

      # Read current values and store backup
      $currentDlls = (Get-ItemProperty -Path $appInitKey -Name "AppInit_DLLs" -ErrorAction SilentlyContinue).AppInit_DLLs
      $currentLoad = (Get-ItemProperty -Path $appInitKey -Name "LoadAppInit_DLLs" -ErrorAction SilentlyContinue).LoadAppInit_DLLs
      $currentSigned = (Get-ItemProperty -Path $appInitKey -Name "RequireSignedAppInit_DLLs" -ErrorAction SilentlyContinue).RequireSignedAppInit_DLLs

      Set-ItemProperty -Path $backupKey -Name "AppInit_DLLs" -Value ($currentDlls ?? "")
      Set-ItemProperty -Path $backupKey -Name "LoadAppInit_DLLs" -Value ($currentLoad ?? 0)
      Set-ItemProperty -Path $backupKey -Name "RequireSignedAppInit_DLLs" -Value ($currentSigned ?? 0)

      # Set new values — derive DLL path from installation directory
      $dllPath = "$InstallDir\dlp_hook_dll.dll"
      $newDlls = if ($currentDlls) { "$currentDlls $dllPath" } else { $dllPath }
      Set-ItemProperty -Path $appInitKey -Name "AppInit_DLLs" -Value $newDlls
      Set-ItemProperty -Path $appInitKey -Name "LoadAppInit_DLLs" -Value 1
      Set-ItemProperty -Path $appInitKey -Name "RequireSignedAppInit_DLLs" -Value 1

      Write-Host "AppInit_DLLs configured for DLP hook DLL fallback"
      ```

   b. Add uninstall/restore logic:
      ```powershell
      # In uninstall section:
      $appInitKey = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Windows"
      $backupKey = "HKLM:\SOFTWARE\DLP\Backup\AppInit_DLLs"

      if (Test-Path $backupKey) {
          $originalDlls = (Get-ItemProperty -Path $backupKey -Name "AppInit_DLLs" -ErrorAction SilentlyContinue).AppInit_DLLs
          $originalLoad = (Get-ItemProperty -Path $backupKey -Name "LoadAppInit_DLLs" -ErrorAction SilentlyContinue).LoadAppInit_DLLs
          $originalSigned = (Get-ItemProperty -Path $backupKey -Name "RequireSignedAppInit_DLLs" -ErrorAction SilentlyContinue).RequireSignedAppInit_DLLs

          if ($originalDlls -ne $null) {
              Set-ItemProperty -Path $appInitKey -Name "AppInit_DLLs" -Value $originalDlls
          } else {
              Remove-ItemProperty -Path $appInitKey -Name "AppInit_DLLs" -ErrorAction SilentlyContinue
          }
          if ($originalLoad -ne $null) {
              Set-ItemProperty -Path $appInitKey -Name "LoadAppInit_DLLs" -Value $originalLoad
          }
          if ($originalSigned -ne $null) {
              Set-ItemProperty -Path $appInitKey -Name "RequireSignedAppInit_DLLs" -Value $originalSigned
          }

          Remove-Item -Path $backupKey -Recurse -Force
          Write-Host "AppInit_DLLs restored from backup"
      }
      ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Installer -> registry | Installer writes HKLM at install time with elevated privileges |
| Periodic task -> process registry | Background tasks read/modify shared DashMap state |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-49-20 | Tampering | Installer registry backup corruption | mitigate | Backup stored in HKLM (admin-only writable). Original values restored on uninstall. |
| T-49-21 | Information Disclosure | Telemetry events expose process counts | accept | Telemetry contains aggregate counts only, not individual PIDs or paths. No PII. |
| T-49-22 | Denial of Service | Periodic backstop sweep under load | accept | Backstop runs every 5 minutes with parallel batches. Minimal CPU impact. |
| T-49-23 | Denial of Service | Cleanup sweep removes active PIDs | mitigate | Cleanup only removes `Exited` and `Failed` states. `Injected` PIDs are never removed by cleanup. |
</threat_model>
