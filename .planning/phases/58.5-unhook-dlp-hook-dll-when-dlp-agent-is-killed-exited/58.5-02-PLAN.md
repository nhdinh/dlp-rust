<objective>
Implement cooperative self-unhook inside the hook DLL: integrate ntdll detour disable and shared-memory unmapping into `UnhookAll()`, add a pipe watchdog thread for agent-crash scenarios, and add an `UnhookCommand` handler that restores hooks, sends an `UnhookAck`, and self-unloads via `FreeLibraryAndExitThread`.

Purpose: This is the in-process side of the unhook protocol. It ensures that when the agent is stopped or killed, every injected process can restore original IAT/ntdll stubs and release shared memory without crashing.
Output: Extended hook DLL with complete self-unhook capability and unit tests verifying UnhookAll state transitions.
</objective>

    Add a public handler:
    ```rust
    pub fn handle_unhook_command(command: &dlp_common::hook_ipc::UnhookCommand) {
        let msg = format!("[dlp-hook] UnhookCommand received: {:?}\0", command.reason);
        debug_log(&msg);

        let pid = std::process::id();
        let result = std::panic::catch_unwind(|| {
            UnhookAll();
        });

        let ack = dlp_common::hook_ipc::UnhookAck {
            pid,
            success: result.is_ok(),
            error: result.err().map(|_| "unhook failed".to_string()),
        };

        crate::pipe_client::send_unhook_ack(crate::DEFAULT_PIPE_NAME, &ack);

        if result.is_ok() {
            self_unload();
        }
    }

    fn self_unload() {
        unsafe {
            let module = windows::Win32::System::LibraryLoader::GetModuleHandleW(w!("dlp_hook_dll.dll"));
            if let Ok(h) = module {
                if !h.is_invalid() {
                    windows::Win32::System::LibraryLoader::FreeLibraryAndExitThread(h, 0);
                }
            }
        }
    }
    ```

    In `dlp-hook-dll/src/pipe_client.rs`, add:
    ```rust
    pub fn send_unhook_ack(pipe_name: &str, ack: &dlp_common::hook_ipc::UnhookAck) -> Result<(), PipeError> {
        let envelope = dlp_common::hook_ipc::IpcEnvelope::V1(dlp_common::hook_ipc::IpcMessageV1 {
            payload: dlp_common::hook_ipc::IpcPayloadV1::UnhookAck(ack.clone()),
        });
        let payload = bincode::serialize(&envelope).map_err(|_| PipeError::Malformed)?;
        send_raw_oneway(pipe_name, &payload)
    }
    ```

    ```rust
    pub fn spawn_watchdog_thread(pipe_name: &'static str) {
        std::thread::spawn(move || {
            const MAX_FAILURES: u32 = 3;
            const POLL_INTERVAL_MS: u64 = 5000;
            let mut failures = 0u32;

            loop {
                std::thread::sleep(Duration::from_millis(POLL_INTERVAL_MS));

                if is_shutting_down() {
                    return;
                }

                if pipe_client::is_pipe_reachable(pipe_name) {
                    failures = 0;
                    continue;
                }

                failures += 1;
                if failures >= MAX_FAILURES {
                    let msg = format!("[dlp-hook] watchdog: agent unreachable after {} failures — self-unhooking\0", MAX_FAILURES);
                    debug_log(&msg);

                    // Emit WatchdogSelfUnload audit event locally via pipe (best-effort).
                    let _ = emit_watchdog_self_unload_alert();

                    handle_unhook_command(&dlp_common::hook_ipc::UnhookCommand {
                        reason: dlp_common::hook_ipc::UnhookReason::AgentCrashDetected,
                        timestamp_secs: std::time::SystemTime::now()
                            .duration_since(std::time::UNIX_EPOCH)
                            .unwrap_or_default()
                            .as_secs(),
                    });
                    return;
                }
            }
        });
    }
    ```

    In `dlp-hook-dll/src/pipe_client.rs`, add:

    ```rust
    pub fn is_pipe_reachable(pipe_name: &str) -> bool {
        match connect_client(pipe_name) {
            Ok(handle) => {
                unsafe { windows::Win32::Foundation::CloseHandle(handle) };
                true
            }
            Err(_) => false,
        }
    }
    ```
