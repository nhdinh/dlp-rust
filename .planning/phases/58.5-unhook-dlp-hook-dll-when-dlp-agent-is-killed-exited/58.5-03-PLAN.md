<objective>
Implement the agent-side unhook dispatch: on graceful shutdown, iterate all `Injected` processes in the `ProcessRegistry`, send `UnhookCommand` via the named pipe to each PID, wait for `UnhookAck`, emit `AgentShutdownUnhook` and `UnhookFailure` audit events, and transition registry entries to `Exited` via `record_unhooked()`.

Purpose: This is the agent-side orchestration of the cooperative unhook protocol. It ensures every injected process receives a clean unhook signal before the agent exits, and the audit trail records the outcome.
Output: Extended agent service with unhook dispatch, registry state transition, and passing unit tests.

        // If multiple entries share the same pid (should not happen in practice),

        for mut entry in self.states.iter_mut() {

    ```rust
    pub fn send_unhook_command(pipe_name: &str, pid: u32, timeout_ms: u64) -> Result<dlp_common::hook_ipc::UnhookAck> {
        let handle = connect_client(pipe_name)?;
        let command = dlp_common::hook_ipc::UnhookCommand {
            reason: dlp_common::hook_ipc::UnhookReason::GracefulShutdown,
            timestamp_secs: std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)
                .unwrap_or_default()
                .as_secs(),
        };
        let envelope = dlp_common::hook_ipc::IpcEnvelope::V1(dlp_common::hook_ipc::IpcMessageV1 {
            payload: dlp_common::hook_ipc::IpcPayloadV1::UnhookCommand(command),
        });
        let payload = bincode::serialize(&envelope).map_err(|e| anyhow::anyhow!("serialize UnhookCommand: {}", e))?;
        write_frame(handle, &payload)?;

        // Read response frame with timeout.
        let frame = read_frame(handle)?;
        close_pipe(handle);

        let response_envelope: dlp_common::hook_ipc::IpcEnvelope = bincode::deserialize(&frame)
            .map_err(|e| anyhow::anyhow!("deserialize UnhookAck: {}", e))?;
        let IpcEnvelope::V1(msg) = response_envelope;
        match msg.payload {
            IpcPayloadV1::UnhookAck(ack) => Ok(ack),
            other => Err(anyhow::anyhow!("expected UnhookAck, got {:?}", other)),
        }
    }
    ```
    Note: `read_frame` is synchronous blocking; for timeout, wrap in a thread or use `tokio::time::timeout` if called from an async context. Since `run_loop_shutdown` is async, use `tokio::task::spawn_blocking` + `tokio::time::timeout`.

    In `dlp-agent/src/service.rs`, add `unhook_all_injected_processes` and call it from `run_loop_shutdown` BEFORE the hook IPC server is stopped (line ~3758):
    ```rust
    async fn unhook_all_injected_processes(
        registry: &Arc<crate::process_registry::ProcessRegistry>,
        pipe_name: &str,
    ) {
        let injected: Vec<u32> = {
            let mut pids = Vec::new();
            for entry in registry.states.iter() {
                if matches!(entry.value().state, crate::process_registry::ProcessState::Injected { .. }) {
                    pids.push(entry.key().pid);
                }
            }
            pids
        };

        if injected.is_empty() {
            return;
        }

        info!(count = injected.len(), "unhooking all injected processes");

        // Emit AgentShutdownUnhook audit event once.
        crate::audit_emitter::emit_unhook_audit(
            dlp_common::EventType::AgentShutdownUnhook,
            std::process::id(),
            true,
            None,
        );

        const UNHOOK_TIMEOUT: Duration = Duration::from_secs(3);
        for pid in injected {
            let result = tokio::time::timeout(UNHOOK_TIMEOUT, tokio::task::spawn_blocking({
                let pipe = pipe_name.to_string();
                move || crate::hook_ipc::send_unhook_command(&pipe, pid, UNHOOK_TIMEOUT.as_millis() as u64)
            })).await;

            match result {
                Ok(Ok(Ok(ack))) => {
                    if ack.success {
                        info!(pid, "unhook acknowledged");
                        registry.record_unhooked_by_pid(pid);
                    } else {
                        warn!(pid, error = ?ack.error, "unhook failed on client side");
                        crate::audit_emitter::emit_unhook_audit(
                            dlp_common::EventType::UnhookFailure,
                            pid,
                            false,
                            ack.error,
                        );
                        registry.record_unhooked_by_pid(pid);
                    }
                }
                Ok(Ok(Err(e))) => {
                    warn!(pid, error = %e, "unhook command failed");
                    crate::audit_emitter::emit_unhook_audit(
                        dlp_common::EventType::UnhookFailure,
                        pid,
                        false,
                        Some(format!("send_unhook_command error: {}", e)),
                    );
                }
                Ok(Err(_)) => {
                    warn!(pid, "unhook command panicked in spawn_blocking");
                    crate::audit_emitter::emit_unhook_audit(
                        dlp_common::EventType::UnhookFailure,
                        pid,
                        false,
                        Some("spawn_blocking panic".to_string()),
                    );
                }
                Err(_) => {
                    warn!(pid, "unhook command timed out");
                    crate::audit_emitter::emit_unhook_audit(
                        dlp_common::EventType::UnhookFailure,
                        pid,
                        false,
                        Some("timeout".to_string()),
                    );
                }
            }
        }
    }
    ```
