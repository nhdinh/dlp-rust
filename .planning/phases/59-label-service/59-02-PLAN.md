           State(state): State<Arc<AppState>>,
           Path(id): Path<String>,

           let row = tokio::task::spawn_blocking(move || {
               LabelRepository::get_by_id(&state.pool, &id)
           }).await.map_err(|e| AppError::Internal(e.to_string()))??;
           
           let old_state = row.label_state.clone();
           let now = chrono::Utc::now().to_rfc3339();

           let updated = tokio::task::spawn_blocking(move || {

               let uow = UnitOfWork::new(&mut conn)?;
               let affected = LabelRepository::update_state(&uow, &id, "expired", &now)?;
               uow.commit()?;
               Ok::<_, AppError>(affected)
           }).await.map_err(|e| AppError::Internal(e.to_string()))??;
           
           if updated == 0 {
               return Err(AppError::NotFound(format!("label {} not found", id)));
           }
           
           state.label_service.invalidate_cache();
           
           // Emit audit event
           if let Err(e) = audit_store::store_events_sync(&state.pool, &[AuditEvent {
               event_type: "label_expire".to_string(),
               label_id: Some(id.clone()),
               old_state: Some(old_state),
               new_state: Some("expired".to_string()),
               path: Some(row.path.clone()),
               tier: Some(row.tier.clone()),
               actor: "admin".to_string(),
               ..Default::default()
           }]) {
               tracing::error!(error = %e, "failed to emit label expire audit event");
           }
           
           // Return updated label
           let updated_row = tokio::task::spawn_blocking(move || {
               LabelRepository::get_by_id(&state.pool, &id)
           }).await.map_err(|e| AppError::Internal(e.to_string()))??;
           
           Ok(Json(LabelResponse::from(updated_row)))
       }

    2. Add `with_mutation` method to `LabelService`:
       ```rust
       pub fn with_mutation<F>(
           &self,
           pool: &Pool,
           mutation: F,
       ) -> Result<MutationContext, AppError>
       where
           F: FnOnce(&UnitOfWork) -> Result<MutationContext, AppError>,
       {
           let mut conn = pool.get().map_err(|e| AppError::Database(e.to_string()))?;
           let uow = UnitOfWork::new(&mut conn).map_err(|e| AppError::Database(e.to_string()))?;
           
           let ctx = mutation(&uow)?;
           
           // Audit emission INSIDE the transaction per D-14
           // If this fails, the transaction rolls back
           let audit_event = AuditEvent {
               event_type: ctx.action.clone(),
               label_id: Some(ctx.label_id.clone()),
               old_state: ctx.old_state.clone(),
               new_state: ctx.new_state.clone(),
               path: Some(ctx.path.clone()),
               tier: Some(ctx.tier.clone()),
               actor: "admin".to_string(),
               ..Default::default()
           };
           if let Err(e) = audit_store::store_events_sync_uow(&uow, &[audit_event]) {
               return Err(AppError::Database(format!("audit emission failed, rolling back: {}", e)));
           }
           
           uow.commit().map_err(|e| AppError::Database(e.to_string()))?;
           
           // Cache invalidation AFTER successful commit per D-15
           self.invalidate_cache();
           
           Ok(ctx)
       }
       ```
