   pub fn emit_label_action(
       pool: &Pool,
       action: &str,
       label_id: &str,
       old_state: Option<&str>,
       new_state: Option<&str>,
       path: &str,
       tier: &str,
       actor: &str,
       reason: Option<&str>,
   ) -> Result<(), rusqlite::Error> {
       let details = format!(
           "label_id={} action={} path={} tier={} old_state={:?} new_state={:?} reason={:?}",
           label_id, action, path, tier, old_state, new_state, reason
       );
       let conn = pool.get().map_err(|e| rusqlite::Error::ToSqlConversionFailure(Box::new(e)))?;
       conn.execute(
           "INSERT INTO audit_events (timestamp, event_type, user_sid, user_name, resource_path, classification, action_attempted, decision, agent_id, session_id) \
            VALUES (?1, 'AdminAction', ?2, ?3, ?4, ?5, ?6, 'ALLOW', 'server', 0)",
           rusqlite::params![
               chrono::Utc::now().to_rfc3339(),
               actor,
               actor,
               details,
               tier,
               action,
           ],
       )?;
       Ok(())
   }

2. `LabelService::with_mutation` helper:
   ```rust
   pub fn with_mutation<F>(
       &self,
       pool: &Pool,
       mutation: F,
   ) -> Result<MutationResult, AppError>
   where
       F: FnOnce(&UnitOfWork) -> Result<MutationResult, AppError>,
   {
       let uow = UnitOfWork::new(pool)?;
       let result = mutation(&uow)?;
       uow.commit()?;
       
       // Cache invalidation after successful commit
       self.invalidate_cache();
       
       // Audit emission (best-effort, logged on failure)
       if let Err(e) = audit_store::emit_label_action(
           pool,
           &result.action,
           &result.label_id,
           result.old_state.as_deref(),
           result.new_state.as_deref(),
           &result.path,
           &result.tier,
           "admin",
           None,
       ) {
           tracing::error!(error = %e, label_id = %result.label_id, "failed to emit label audit event");
       }
       
       Ok(result)
   }
   ```
