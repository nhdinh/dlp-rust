<objective>
Integrate label-aware ABAC evaluation with performance-optimized flag caching, fail-closed error handling, rich resolution results, and comprehensive audit logging. When the `label_aware_evaluation_enabled` system flag is true, the PolicyStore resolves `Resource.classification` from the LabelService instead of using the request's hardcoded classification. Unlabeled resources fall back to `UnclassifiedBlocked`.

Purpose: This is the enforcement-time integration that makes labels meaningful. Without it, labels are just metadata. With it, the ABAC engine uses the central label database for tier decisions.
Output: Label-aware ABAC evaluation gated by a cached system flag (default off), with fail-closed semantics and audit trail.
</objective>

   ```rust
   let mut resource = ctx.resource.clone();
   let mut classification_overridden = false;
   let mut override_source: Option<String> = None;
   
   // Label-aware evaluation: resolve tier from LabelService when enabled
   if label_aware_enabled {
       if let Some(service) = label_service {
           if let Some(ref path) = ctx.resource_path {
               self.label_eval_count.fetch_add(1, Ordering::Relaxed);
               match service.resolve_tier(path) {
                   ResolvedTier::Exact { tier } => {
                       if let Some(classification) = tier.to_classification() {
                           resource.classification = classification;
                           classification_overridden = true;
                           override_source = Some(format!("exact label: {}", path));
                       } else {
                           // UnclassifiedBlocked: deny-all (T4 is the deny-all tier in this system)
                           resource.classification = Classification::T4;
                           classification_overridden = true;
                           override_source = Some(format!("exact label UnclassifiedBlocked: {}", path));
                       }
                   }
                   ResolvedTier::Inherited { tier, parent_path } => {
                       if let Some(classification) = tier.to_classification() {
                           resource.classification = classification;
                           classification_overridden = true;
                           override_source = Some(format!("inherited from parent: {}", parent_path));
                       } else {
                           resource.classification = Classification::T4;
                           classification_overridden = true;
                           override_source = Some(format!("inherited UnclassifiedBlocked from parent: {}", parent_path));
                       }
                   }
                   ResolvedTier::FallbackUnclassifiedBlocked => {
                       self.label_fallback_count.fetch_add(1, Ordering::Relaxed);
                       resource.classification = Classification::T4;
                       classification_overridden = true;
                       override_source = Some(format!("no label found for path: {}", path));
                   }
                   ResolvedTier::LookupFailed => {
                       self.label_lookup_failed_count.fetch_add(1, Ordering::Relaxed);
                       tracing::error!(path, "label resolution failed — denying access (fail-closed)");
                       resource.classification = Classification::T4;
                       classification_overridden = true;
                       override_source = Some(format!("label lookup failed for path: {}", path));
                   }
               }
               
               // Audit log when classification is overridden
               if classification_overridden {
                   tracing::info!(
                       original = ?ctx.resource.classification,
                       resolved = ?resource.classification,
                       source = ?override_source,
                       "classification overridden by label service"
                   );
               }
           } else {
               // Flag is ON but no resource_path provided — fail-closed
               resource.classification = Classification::T4;
               classification_overridden = true;
               override_source = Some("label-aware enabled but no resource_path provided".to_string());
               tracing::info!(
                   original = ?ctx.resource.classification,
                   resolved = ?resource.classification,
                   "classification overridden: label-aware enabled but no resource_path"
               );
           }
       }
   }
   ```
