
    4. Modify `dlp-agent/src/server_client.rs`:
       - Find the heartbeat success handler (where the agent knows the server is reachable).
       - Add drain logic:
         ```rust
         if offline_audit_queue::try_acquire_drain_lock() {
             match offline_audit_queue::count(&conn) {
                 Ok(0) | Ok(0..) if count == 0 => {},
                 Ok(n) if n > 0 => {
                     match offline_audit_queue::drain(&conn, 100) {
                         Ok(events) => {
                             // Forward events to server
                             let ids: Vec<i64> = events.iter().map(|(id, _)| *id).collect();
                             let json_events: Vec<String> = events.into_iter().map(|(_, json)| json).collect();
                             match forward_events_to_server(&json_events).await {
                                 Ok(()) => {
                                     let _ = offline_audit_queue::delete(&conn, &ids);
                                     tracing::info!(count = ids.len(), "Drained queued events to server");
                                 }
                                 Err(e) => {
                                     tracing::warn!(error = %e, "Failed to forward drained events");
                                 }
                             }
                         }
                         Err(e) => tracing::warn!(error = %e, "Drain failed"),
                     }
                 }
                 _ => {}
             }
             offline_audit_queue::release_drain_lock();
         }
         ```
       - The `forward_events_to_server` should use the existing HTTP client / server client mechanism.

       ```rust
           Ok(()) => {},

               let overflow = AuditEvent {

                   session_id: 0,
                   classification: Classification::default(),
                   ..Default::default()
               };
               // Emit the synthetic event (this one goes to JSONL only, not re-queued)
               let _ = write_jsonl(&overflow);
           }
           Err(e) => tracing::warn!(error = %e, "Queue enqueue failed"),
       }
       ```
