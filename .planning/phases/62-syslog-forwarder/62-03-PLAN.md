       ```rust
       //! Agent-side syslog offline queue.
       //!
       //! Stores audit events locally when the dlp-server is unreachable.
       //! Events are encrypted with DPAPI (Windows only) before writing to SQLite.
       //! On heartbeat success, the agent drains the queue and forwards events.

       use rusqlite::{params, Connection};

       #[cfg(windows)]
       use dlp_server::crypto::{dpapi_protect, dpapi_unprotect};

       /// Initialize the agent_syslog_queue table on the given connection.
       pub fn init_table(conn: &Connection) -> Result<(), rusqlite::Error> {
           conn.execute_batch(
               "CREATE TABLE IF NOT EXISTS agent_syslog_queue (
                   id             INTEGER PRIMARY KEY AUTOINCREMENT,
                   event_json_dpapi BLOB NOT NULL,
                   created_at     TEXT NOT NULL,
                   retry_count    INTEGER NOT NULL DEFAULT 0
               );
               CREATE INDEX IF NOT EXISTS idx_agent_syslog_queue_created_at
                   ON agent_syslog_queue(created_at);"
           )
       }

       /// Enqueue an event. On Windows, encrypts with DPAPI before storing.
       /// On non-Windows, stores plaintext (DPAPI is Windows-only).
       pub fn enqueue(conn: &Connection, event_json: &str) -> Result<(), SyslogQueueError> {
           #[cfg(windows)]
           let blob = dpapi_protect(event_json.as_bytes())
               .map_err(|e| SyslogQueueError::Encrypt(e.to_string()))?;
           #[cfg(not(windows))]
           let blob = event_json.as_bytes().to_vec();

           conn.execute(
               "INSERT INTO agent_syslog_queue (event_json_dpapi, created_at, retry_count)
                VALUES (?1, ?2, 0)",
               params![blob, chrono::Utc::now().to_rfc3339()],
           )?;
           Ok(())
       }

       /// Drain oldest N events from the queue. Returns (id, decrypted_json) tuples.
       pub fn drain(conn: &Connection, batch_size: usize) -> Result<Vec<(i64, String)>, SyslogQueueError> {
           let mut stmt = conn.prepare(
               "SELECT id, event_json_dpapi FROM agent_syslog_queue
                ORDER BY created_at LIMIT ?1"
           )?;
           let rows = stmt.query_map([batch_size], |row| {
               let id: i64 = row.get(0)?;
               let blob: Vec<u8> = row.get(1)?;
               #[cfg(windows)]
               let plaintext = dpapi_unprotect(&blob)
                   .map_err(|e| rusqlite::Error::InvalidParameterName(format!("dpapi: {e}")))?;
               #[cfg(not(windows))]
               let plaintext = blob;
               let json = String::from_utf8(plaintext)
                   .map_err(|e| rusqlite::Error::InvalidParameterName(format!("utf8: {e}")))?;
               Ok((id, json))
           })?;
           rows.collect::<Result<Vec<_>, _>>()
               .map_err(SyslogQueueError::from)
       }

       /// Delete rows by id after successful forwarding.
       pub fn delete(conn: &Connection, ids: &[i64]) -> Result<(), rusqlite::Error> {
           if ids.is_empty() { return Ok(()); }
           let placeholders: Vec<String> = ids.iter().map(|_| "?".to_string()).collect();
           let sql = format!("DELETE FROM agent_syslog_queue WHERE id IN ({})", placeholders.join(","));
           let mut stmt = conn.prepare(&sql)?;
           let params: Vec<&dyn rusqlite::ToSql> = ids.iter().map(|id| id as &dyn rusqlite::ToSql).collect();
           stmt.execute(rusqlite::params_from_iter(params.iter()))?;
           Ok(())
       }

       /// Return current queue depth.
       pub fn count(conn: &Connection) -> Result<i64, rusqlite::Error> {
           conn.query_row("SELECT COUNT(*) FROM agent_syslog_queue", [], |row| row.get(0))
       }

       /// Enforce max queue size with FIFO tail-drop (per D-08, D-09).
       /// If count exceeds max_size, delete newest rows until count == max_size.
       pub fn enforce_tail_drop(conn: &Connection, max_size: i64) -> Result<(), rusqlite::Error> {
           let count: i64 = conn.query_row("SELECT COUNT(*) FROM agent_syslog_queue", [], |row| row.get(0))?;
           if count > max_size {
               let excess = count - max_size;
               conn.execute(
                   "DELETE FROM agent_syslog_queue WHERE id IN (
                       SELECT id FROM agent_syslog_queue ORDER BY created_at DESC LIMIT ?1
                   )",
                   params![excess],
               )?;
           }
           Ok(())
       }

       #[derive(Debug, thiserror::Error)]
       pub enum SyslogQueueError {
           #[error("DPAPI encrypt error: {0}")]
           Encrypt(String),
           #[error("DPAPI decrypt error: {0}")]
           Decrypt(String),
           #[error("database error: {0}")]
           Database(#[from] rusqlite::Error),
       }
       ```
       - Pitfall 4 (RESEARCH.md): DPAPI data is lost on machine rebuild. Document this in module doc comment.
       - `enforce_tail_drop` implements FIFO tail-drop per D-08. Called after every enqueue.
       - Default max_size: 10,000 events per D-09.
       - CRITICAL: The DELETE subquery uses `ORDER BY created_at DESC` to delete newest events (tail-drop), matching D-08. The previous `ASC` was incorrect (would delete oldest = head-drop).

       ```rust
       //! Syslog forwarder configuration screen.
       //!
       //! Mirrors the SiemConfig screen pattern exactly.

       use crossterm::event::{KeyCode, KeyEvent};
       use ratatui::layout::Rect;
       use ratatui::Frame;

       use crate::app::{App, Screen, StatusKind};

       const SYSLOG_KEYS: [&str; 13] = [
           "host", "port", "enabled", "protocol", "facility_code", "format",
           "batching_enabled", "severity_alert", "severity_block", "severity_audit",
           "queue_policy", "queue_max_size", "tls_min_version",
       ];
       const SYSLOG_TEST_ROW: usize = 13;
       const SYSLOG_SAVE_ROW: usize = 14;
       const SYSLOG_BACK_ROW: usize = 15;
       pub const SYSLOG_ROW_COUNT: usize = 16;

       // Picker options for select fields
       const PROTOCOL_OPTIONS: [&str; 1] = ["tls"];
       const FACILITY_OPTIONS: [&str; 8] = [
           "LOCAL0", "LOCAL1", "LOCAL2", "LOCAL3", "LOCAL4",
           "LOCAL5", "LOCAL6", "LOCAL7",
       ];
       const FORMAT_OPTIONS: [&str; 1] = ["json"];
       const QUEUE_POLICY_OPTIONS: [&str; 3] = [
           "fifo_tail_drop", "fifo_head_drop", "ring_buffer",
       ];
       const TLS_VERSION_OPTIONS: [&str; 2] = ["1.2", "1.3"];

       pub fn handle_syslog_config(app: &mut App, key: KeyEvent) {
           let (selected, editing) = match &app.screen {
               Screen::SyslogConfig { selected, editing, .. } => (*selected, *editing),
               _ => return,
           };

           if editing {
               handle_syslog_config_editing(app, key, selected);
           } else {
               handle_syslog_config_nav(app, key, selected);
           }
       }

       fn handle_syslog_config_nav(app: &mut App, key: KeyEvent, selected: usize) {
           match key.code {
               KeyCode::Up => {
                   let new = selected.saturating_sub(1);
                   if let Screen::SyslogConfig { config, editing: _, buffer: _, .. } = &mut app.screen {
                       app.screen = Screen::SyslogConfig {
                           config: config.clone(),
                           selected: new,
                           editing: false,
                           buffer: String::new(),
                       };
                   }
               }
               KeyCode::Down => {
                   let new = (selected + 1).min(SYSLOG_ROW_COUNT - 1);
                   if let Screen::SyslogConfig { config, editing: _, buffer: _, .. } = &mut app.screen {
                       app.screen = Screen::SyslogConfig {
                           config: config.clone(),
                           selected: new,
                           editing: false,
                           buffer: String::new(),
                       };
                   }
               }
               KeyCode::Enter => {
                   match selected {
                       SYSLOG_TEST_ROW => action_test_syslog_config(app),
                       SYSLOG_SAVE_ROW => action_save_syslog_config(app),
                       SYSLOG_BACK_ROW => app.screen = Screen::SystemMenu { selected: 0 },
                       _ => {
                           // Enter edit mode for the selected field
                           if let Screen::SyslogConfig { config, selected: _, editing: _, buffer: _ } = &app.screen {
                               let key_str = SYSLOG_KEYS[selected];
                               let current = config.get(key_str).and_then(|v| v.as_str()).unwrap_or("");
                               app.screen = Screen::SyslogConfig {
                                   config: config.clone(),
                                   selected,
                                   editing: true,
                                   buffer: current.to_string(),
                               };
                           }
                       }
                   }
               }
               KeyCode::Esc => {
                   app.screen = Screen::SystemMenu { selected: 0 };
               }
               _ => {}
           }
       }

       fn handle_syslog_config_editing(app: &mut App, key: KeyEvent, selected: usize) {
           match key.code {
               KeyCode::Enter => {
                   // Commit buffer to config
                   if let Screen::SyslogConfig { config, selected: _, editing: _, buffer } = &app.screen {
                       let mut new_config = config.clone();
                       let key_str = SYSLOG_KEYS[selected];
                       let value = match key_str {
                           "port" | "facility_code" | "severity_alert" | "severity_block" | "severity_audit" | "queue_max_size" => {
                               serde_json::json!(buffer.parse::<i64>().unwrap_or_default())
                           }
                           "enabled" | "batching_enabled" => {
                               serde_json::json!(buffer == "true" || buffer == "1")
                           }
                           _ => serde_json::json!(buffer.clone()),
                       };
                       if let Some(obj) = new_config.as_object_mut() {
                           obj.insert(key_str.to_string(), value);
                       }
                       app.screen = Screen::SyslogConfig {
                           config: new_config,
                           selected,
                           editing: false,
                           buffer: String::new(),
                       };
                   }
               }
               KeyCode::Esc => {
                   // Cancel editing
                   if let Screen::SyslogConfig { config, selected: _, editing: _, buffer: _ } = &app.screen {
                       app.screen = Screen::SyslogConfig {
                           config: config.clone(),
                           selected,
                           editing: false,
                           buffer: String::new(),
                       };
                   }
               }
               KeyCode::Char(c) => {
                   if let Screen::SyslogConfig { config, selected: _, editing: _, ref mut buffer } = &mut app.screen {
                       buffer.push(c);
                   }
               }
               KeyCode::Backspace => {
                   if let Screen::SyslogConfig { config, selected: _, editing: _, ref mut buffer } = &mut app.screen {
                       buffer.pop();
                   }
               }
               _ => {}
           }
       }

       pub fn action_load_syslog_config(app: &mut App) {
           match app.rt.block_on(app.client.get::<serde_json::Value>("admin/syslog-config")) {
               Ok(config) => {
                   app.screen = Screen::SyslogConfig {
                       config,
                       selected: 0,
                       editing: false,
                       buffer: String::new(),
                   };
               }
               Err(e) => app.set_status(format!("Failed to load syslog config: {e}"), StatusKind::Error),
           }
       }

       pub fn action_save_syslog_config(app: &mut App) {
           let payload = match &app.screen {
               Screen::SyslogConfig { config, .. } => config.clone(),
               _ => return,
           };
           match app.rt.block_on(app.client.put::<serde_json::Value, _>("admin/syslog-config", &payload)) {
               Ok(_) => app.set_status("Syslog config saved", StatusKind::Success),
               Err(e) => app.set_status(format!("Failed to save: {e}"), StatusKind::Error),
           }
       }

       pub fn action_test_syslog_config(app: &mut App) {
           match app.rt.block_on(app.client.post::<serde_json::Value, _>("admin/syslog-config/test", &serde_json::json!({}))) {
               Ok(response) => {
                   let status = response.get("status").and_then(|v| v.as_str()).unwrap_or("unknown");
                   let message = response.get("message").and_then(|v| v.as_str()).unwrap_or("");
                   if status == "ok" {
                       app.set_status(format!("Test OK: {message}"), StatusKind::Success);
                   } else {
                       app.set_status(format!("Test failed: {message}"), StatusKind::Error);
                   }
               }
               Err(e) => app.set_status(format!("Test request failed: {e}"), StatusKind::Error),
           }
       }

       pub fn draw_syslog_config(frame: &mut Frame, area: Rect, config: &serde_json::Value, selected: usize, editing: bool, buffer: &str) {
           use ratatui::widgets::{Block, Borders, List, ListItem};
           use ratatui::style::{Style, Modifier};

           let block = Block::default()
               .title("Syslog Forwarder Configuration")
               .borders(Borders::ALL);

           let mut items: Vec<ListItem> = Vec::with_capacity(SYSLOG_ROW_COUNT);

           for (idx, key) in SYSLOG_KEYS.iter().enumerate() {
               let value = format_config_field_value(config, key);
               let label = format_field_label(key);
               let text = if idx == selected && editing {
                   format!("{label}: [{buffer}]")
               } else {
                   format!("{label}: {value}")
               };
               let style = if idx == selected {
                   Style::default().add_modifier(Modifier::REVERSED)
               } else {
                   Style::default()
               };
               items.push(ListItem::new(text).style(style));
           }

           // Test row
           let test_style = if selected == SYSLOG_TEST_ROW {
               Style::default().add_modifier(Modifier::REVERSED)
           } else {
               Style::default()
           };
           items.push(ListItem::new("[ Test Connection ]").style(test_style));

           // Save row
           let save_style = if selected == SYSLOG_SAVE_ROW {
               Style::default().add_modifier(Modifier::REVERSED)
           } else {
               Style::default()
           };
           items.push(ListItem::new("[ Save ]").style(save_style));

           // Back row
           let back_style = if selected == SYSLOG_BACK_ROW {
               Style::default().add_modifier(Modifier::REVERSED)
           } else {
               Style::default()
           };
           items.push(ListItem::new("[ Back ]").style(back_style));

           let list = List::new(items).block(block);
       ```
