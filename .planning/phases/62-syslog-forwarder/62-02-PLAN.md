       async fn update_syslog_config_handler(
           State(state): State<Arc<AppState>>,
           Json(payload): Json<SyslogConfigPayload>,
       ) -> Result<Json<SyslogConfigPayload>, AppError> {
           // Validation
           if payload.port < 1 || payload.port > 65535 {
               return Err(AppError::BadRequest("port must be 1-65535".to_string()));
           }
           if payload.facility_code < 16 || payload.facility_code > 23 {
               return Err(AppError::BadRequest("facility_code must be 16-23 (LOCAL0-LOCAL7)".to_string()));
           }
           let valid_policies = ["fifo_tail_drop", "fifo_head_drop", "ring_buffer"];
           if !valid_policies.contains(&payload.queue_policy.as_str()) {
               return Err(AppError::BadRequest(format!(
                   "queue_policy must be one of: {}",
                   valid_policies.join(", ")
               )));
           }

           let now = chrono::Utc::now().to_rfc3339();
           let p = payload.clone();
           let pool = Arc::clone(&state.pool);
           let crypto = Arc::clone(&state.crypto);

           tokio::task::spawn_blocking(move || -> Result<(), AppError> {
               let mut conn = pool.get().map_err(AppError::from)?;
               let uow = db::UnitOfWork::new(&mut conn).map_err(AppError::Database)?;
               let record = SyslogConfigRow {

                   port: p.port,
                   enabled: if p.enabled { 1 } else { 0 },
                   protocol: p.protocol,
                   facility_code: p.facility_code,
                   format: p.format,
                   batching_enabled: if p.batching_enabled { 1 } else { 0 },
                   severity_alert: p.severity_alert,
                   severity_block: p.severity_block,

                   queue_policy: p.queue_policy,

                   tls_min_version: p.tls_min_version,
                   updated_at: now,
               };

               uow.commit().map_err(AppError::Database)?;
               Ok(())
           })

    1. In `dlp-server/src/main.rs`, after `AppState` construction and before the axum server starts, spawn a background drain loop task:
       ```rust
       // Spawn background syslog queue drain loop (per D-06, RESEARCH.md Q3)
       let drain_syslog = syslog.clone();
       let drain_pool = Arc::clone(&pool);
       let drain_crypto = Arc::clone(&crypto);
       tokio::spawn(async move {
           let interval_secs = std::env::var("SYSLOG_DRAIN_INTERVAL_SECS")
               .ok()
               .and_then(|s| s.parse().ok())
               .unwrap_or(30u64);
           let mut interval = tokio::time::interval(std::time::Duration::from_secs(interval_secs));
           interval.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);

           let mut consecutive_failures: u32 = 0;

           loop {
               interval.tick().await;

               // Check if there are queued events
               let count = match tokio::task::spawn_blocking({
                   let pool = Arc::clone(&drain_pool);
                   move || SyslogQueueRepository::count(&pool)
               }).await {
                   Ok(Ok(c)) => c,
                   Ok(Err(e)) => {
                       tracing::warn!(error = %e, "syslog drain: failed to count queue");
                       consecutive_failures += 1;
                       continue;
                   }
                   Err(e) => {
                       tracing::warn!(error = %e, "syslog drain: join error counting queue");
                       consecutive_failures += 1;
                       continue;
                   }
               };

               if count == 0 {
                   consecutive_failures = 0; // reset backoff when queue is empty
                   continue;
               }

               // Drain a batch
               let batch = match tokio::task::spawn_blocking({
                   let pool = Arc::clone(&drain_pool);
                   let crypto = Arc::clone(&drain_crypto);
                   move || SyslogQueueRepository::drain(&pool, &crypto, 100)
               }).await {
                   Ok(Ok(b)) => b,
                   Ok(Err(e)) => {
                       tracing::warn!(error = %e, "syslog drain: failed to read batch");
                       consecutive_failures += 1;
                       continue;
                   }
                   Err(e) => {
                       tracing::warn!(error = %e, "syslog drain: join error reading batch");
                       consecutive_failures += 1;
                       continue;
                   }
               };

               if batch.is_empty() {
                   consecutive_failures = 0;
                   continue;
               }

               // Deserialize and forward
               let events: Vec<AuditEvent> = match batch.iter()
                   .map(|(_, json)| serde_json::from_str::<AuditEvent>(json))
                   .collect::<Result<Vec<_>, _>>() {
                   Ok(e) => e,
                   Err(e) => {
                       tracing::error!(error = %e, "syslog drain: failed to deserialize queued events");
                       // Don't increment consecutive_failures for deserialization errors --
                       // these are data corruption issues, not network issues.
                       // Log and skip this batch (they'll be retried indefinitely or until
                       // a max retry limit is added in a future phase).
                       continue;
                   }
               };

               match drain_syslog.forward(&events).await {
                   Ok(()) => {
                       // Delete successfully forwarded events
                       let ids: Vec<i64> = batch.iter().map(|(id, _)| *id).collect();
                       if let Err(e) = tokio::task::spawn_blocking({
                           let pool = Arc::clone(&drain_pool);
                           move || {
                               let mut conn = pool.get()?;
                               let uow = db::UnitOfWork::new(&mut conn)?;
                               SyslogQueueRepository::delete(&uow, &ids)?;
                               uow.commit()?;
                               Ok::<_, AppError>(())
                           }
                       }).await {
                           tracing::warn!(error = %e, "syslog drain: failed to delete forwarded events");
                           // Events were forwarded but not deleted -- they'll be re-forwarded
                           // on next drain (at-least-once semantics). This is acceptable.
                       }
                       tracing::info!(count = events.len(), "syslog drain: forwarded queued events");
                       consecutive_failures = 0;
                   }
                   Err(e) => {
                       tracing::warn!(error = %e, count = events.len(), "syslog drain: forward failed");
                       consecutive_failures += 1;
                   }
               }

               // Exponential backoff on failure
               if consecutive_failures > 0 {
                   let base = 1u64;
                   let max = 60u64;
                   let exp = std::cmp::min(consecutive_failures, 6); // cap at 64s
                   let delay = std::cmp::min(base * 2u64.pow(exp), max);
                   let jitter = rand::random::<u64>() % (delay / 2 + 1);
                   let backoff = std::time::Duration::from_secs(delay + jitter);
                   tracing::info!(backoff_secs = backoff.as_secs(), "syslog drain: backing off");
                   tokio::time::sleep(backoff).await;
               }
           }
       });
       ```
       - The loop uses `tokio::time::interval` with `MissedTickBehavior::Skip` to avoid burst-draining on recovery.
       - Batch size of 100 balances throughput with memory usage.
       - `consecutive_failures` tracks backoff state; reset to 0 on any success or empty queue.
       - Backoff formula: min(2^failures, 60) seconds + jitter up to 50%.
       - Deserialization errors are treated separately from network errors (no backoff increment).
       - Delete-after-forward is best-effort; duplicate forwarding on next drain is acceptable (at-least-once).
