       ```
       let handler = Arc::new(move |req: HookRequest| {
           let current_version = cache_ref.lock().current_version();
           
           // Check if DLL is stale.
           if req.cache_version > 0 && req.cache_version < current_version {
               // DLL has stale cache. Still process the request, but the response
               // will include the current version so the DLL can detect the update.
               // (Per D-16, no pipe broadcast needed — DLL detects via version word poll.)
           }
           
           // Existing classification logic (from prior phases).
           let decision = classify_request(&req.path, &req.action, &policy_store
           );
           
           // Build cache hint for cache warming.
           let cache_hint = if let Some(classification) = classify_path_only(&req.path) {
               let ttl_secs = match classification {
                   Classification::T4 => 30,
                   Classification::T3 => 60,
                   Classification::T2 => 300,
                   Classification::T1 => 1800,
               };
               Some((req.path.clone(), classification as u8, ttl_secs))
           } else {
               None
           };
           
           HookResponse {
               decision,
               reason: "classified".to_string(),
               cache_hint,
               cache_version: current_version,
           }
       });
       ```

       ```
       pub(crate) fn classify_path(
           path: &str,
           action: &str,
           pipe_name: &str,
       ) -> Result<Decision, pipe_client::PipeError> {
           let req = HookRequest {
               path: path.to_string(),
               action: action.to_string(),
               pid: std::process::id(),
               tid: get_current_thread_id(),
               file_object: 0,
               journal_seq: 0,
               op: action_to_hook_op(action),
               cache_version: get_last_seen_cache_version(),
           };
           let resp = pipe_client::send_request(pipe_name, &req, 50)?;
           
           // Warm thread-local LRU with cache hint.
           if let Some((hint_path, tier, ttl_secs)) = resp.cache_hint {
               if let Ok(classification) = Classification::try_from(tier) {
                   classification_cache::thread_local_lru_insert(
                       &hint_path,
                       classification,
                       ttl_secs,
                   );
               }
           }
           
           // Update last seen cache version.
           if resp.cache_version > 0 {
               update_last_seen_cache_version(resp.cache_version);
           }
           
           Ok(resp.decision)
       }
       ```
