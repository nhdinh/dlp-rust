       struct TelemetryState {
           call_count: u32,
           cache_hits: u32,
           cache_misses: u32,
           pipe_roundtrips: u32,

       }
       ```

    3. **Public API** (all inline for zero overhead):
       - `record_cache_hit(latency_qpc: i64)`:
         - Convert QPC delta to microseconds using cached frequency
         - Increment cache_hits, call_count
         - Add to histogram bucket
         - If call_count >= TELEMETRY_EMIT_INTERVAL: call emit_telemetry()
       - `record_cache_miss(latency_qpc: i64)`:
         - Same as hit but increments cache_misses
       - `record_pipe_roundtrip(latency_qpc: i64)`:
         - Increments pipe_roundtrips
         - Adds to histogram (separate from cache lookups)
       - `emit_telemetry()`:
         - Serialize telemetry into pipe buffer
         - Send via `pipe_client::send_raw_request` to a telemetry pipe (or reuse main pipe with a special message type)

       - `fn qpc_to_us(delta: i64, freq: i64) -> u64 { (delta as u64 * 1_000_000) / freq as u64 }`
       - If freq is 0 or negative, fall back to `Instant::now()` elapsed microseconds

    5. **Histogram bucket selection**:
       - `fn bucket_index(latency_us: u64) -> usize`:

       fn classify_and_log_path(path: &str, action: &str, fn_name: &str, deny_return: DenyReturn) -> Option<DenyReturn> {
           // Deferred background thread spawn.
           if !BG_THREAD_SPAWNED.swap(true, Ordering::SeqCst) {
               crate::background_thread::spawn_background_thread();
           }

           let path_hash = crate::hash_path(path);
           let start_qpc = unsafe { query_performance_counter() };
           let is_write = is_write_action(action);

           // 1. ALLOWLIST CHECK (fastest path).
           if crate::allowlist::is_allowed(path) {
               let latency_qpc = unsafe { query_performance_counter() } - start_qpc;
               crate::perf_telemetry::record_cache_hit(latency_qpc); // Treat as instant "hit"
               let msg = format!("[dlp-hook] ALLOW(allowlist) {} hash={:016x}\0", fn_name, path_hash);
               crate::debug_log(&msg);
               return None;
           }

           // 2. CACHE LOOKUP.
           let cache_result = crate::classification_cache::lookup(path);
           let cache_latency_qpc = unsafe { query_performance_counter() } - start_qpc;

           match cache_result {
               Some(classification) => {
                   crate::perf_telemetry::record_cache_hit(cache_latency_qpc);
                   if classification.is_sensitive() && is_write {
                       let msg = format!("[dlp-hook] DENY(cache) {} hash={:016x} latency={}us\0", ...);
                       crate::debug_log(&msg);
                       return Some(deny_return);
                   }
                   let msg = format!("[dlp-hook] ALLOW(cache) {} hash={:016x} latency={}us\0", ...);
                   crate::debug_log(&msg);
                   return None;
               }
               None => {
                   crate::perf_telemetry::record_cache_miss(cache_latency_qpc);
               }
           }

           // 3. BUILD-TOOL BYPASS (skip pipe for build tools, use asymmetric decision).
           if crate::allowlist::is_build_tool() {
               let msg = format!("[dlp-hook] ALLOW(build-tool) {} hash={:016x}\0", fn_name, path_hash);
               crate::debug_log(&msg);
               return None; // Build tools are trusted; cache miss = allow
           }

           // 4. FAIL-MODE + PIPE (from Plan 04).
           // ... existing fail-mode and pipe logic ...
       }
       ```
