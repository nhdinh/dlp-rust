       - `pub fn new(db_path: &Path) -> Result<Self, DaclStagingError>` — opens connection, calls init_staging_table
       - `pub fn stage_removal(&self, path: &str) -> Result<(), DaclStagingError>` — INSERT OR REPLACE a 'remove' row with staged_at=now, applied_at=NULL
       - `pub fn stage_add(&self, path: &str) -> Result<(), DaclStagingError>` — INSERT OR REPLACE an 'add' row with staged_at=now, applied_at=NULL
       - `pub fn mark_applied(&self, path: &str) -> Result<(), DaclStagingError>` — UPDATE applied_at=now WHERE path=?
       - `pub fn is_staged(&self, path: &str) -> Result<bool, DaclStagingError>` — returns true if any row exists for path (regardless of applied_at)
       - `pub fn is_staged_and_applied(&self, path: &str) -> Result<bool, DaclStagingError>` — returns true if row exists with applied_at NOT NULL
       - `pub fn gc_expired_rows(&self, ttl_minutes: i64) -> Result<usize, DaclStagingError>` — DELETE WHERE applied_at IS NOT NULL AND datetime(staged_at, '+{ttl} minutes') < datetime('now')
       - `pub fn list_all(&self) -> Result<Vec<StagingRow>, DaclStagingError>` — SELECT * FROM protected_paths_staging
       - `pub fn get_row(&self, path: &str) -> Result<Option<StagingRow>, DaclStagingError>` — SELECT single row by path

    5. **Free function for batch staging**:
       ```rust
       pub fn stage_removals(db: &std::sync::Mutex<rusqlite::Connection>, paths: &[String]) -> Result<usize, DaclStagingError> {
           let conn = db.lock().map_err(|_| DaclStagingError::Sqlite(rusqlite::Error::InvalidPath(std::path::PathBuf::from("mutex poisoned"))))?;
           let mut count = 0;
           for path in paths {
               conn.execute(
                   "INSERT OR REPLACE INTO protected_paths_staging (path, operation, staged_at, applied_at) VALUES (?1, 'remove', ?2, NULL)",
                   (path, Utc::now().to_rfc3339()),
               )?;
               count += 1;
           }
           Ok(count)
       }
       ```
       This function is called by `apply_payload_to_config` (Plan 52-07) to stage removals without needing a `DaclStaging` instance.

    6. **GC task spawner**:
       ```rust
       pub fn spawn_gc_task(
           staging: Arc<DaclStaging>,
           interval_secs: u64,
           mut shutdown_rx: tokio::sync::watch::Receiver<bool>,
       ) -> tokio::task::JoinHandle<()> {
           tokio::spawn(async move {
               let mut interval = tokio::time::interval(Duration::from_secs(interval_secs));
               interval.tick().await; // consume immediate first tick
               loop {
                   tokio::select! {
                       _ = interval.tick() => {
                           match staging.gc_expired_rows(5) {
                               Ok(count) => {
                                   if count > 0 {
                                       tracing::info!(deleted = count, "staging GC removed expired rows");
                                   }
                               }
                               Err(e) => tracing::warn!(error = %e, "staging GC failed"),
                           }
                       }
                       _ = shutdown_rx.changed() => {
                           tracing::info!("staging GC task shutting down");
                           return;
                       }
                   }
               }
           })
       }
       ```
