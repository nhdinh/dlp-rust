       ```rust
       impl ApprovalCacheKey {
           pub fn from_evaluation(
               response: &EvaluateResponse,
               sid: &str,
               action: &str,
               dst: Option<&str>,
           ) -> Option<Self> {
               let obj_id = response.matched_label_id.as_deref()?;
               Some(Self::new(sid, obj_id, action, dst))
           }
       }
       ```

    2. `check_approval_override` — free function that wraps the cache check:
       ```rust
       pub fn check_approval_override(
           cache: &ApprovalCache,
           response: &EvaluateResponse,
           sid: &str,
           action: &str,
           dst: Option<&str>,
       ) -> Option<EvaluateResponse> {
           let key = ApprovalCacheKey::from_evaluation(response, sid, action, dst)?;
           cache.check(&key, dst)
       }
       ```

      - `test_approval_cache_key_from_evaluation_some` — key constructed when matched_label_id is Some
