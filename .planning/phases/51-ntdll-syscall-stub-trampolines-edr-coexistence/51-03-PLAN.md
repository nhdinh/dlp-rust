pub(crate) unsafe fn extract_nt_path(objectattributes: *mut std::ffi::c_void) -> String {
    if objectattributes.is_null() { return String::new(); }
    let object_name_ptr = *(objectattributes.offset(OBJECT_ATTRIBUTES_OBJECT_NAME_OFFSET) as *mut *mut u8);
    if object_name_ptr.is_null() { return String::new(); }
    let buffer = *(object_name_ptr.offset(UNICODE_STRING_BUFFER_OFFSET) as *mut *mut u16);
    let length = *(object_name_ptr as *const u16) as usize;
    if buffer.is_null() || length == 0 { return String::new(); }
    let chars = (length / 2).min(MAX_WIDE_CHARS);
    let slice = std::slice::from_raw_parts(buffer, chars);
    String::from_utf16_lossy(slice)
}
```

    #[unsafe(no_mangle)]
    pub unsafe extern "system" fn NtdllTrampolineNtCreateFile(...) -> NTSTATUS {
        crate::crash_guard::guard_trampoline(
            "NtCreateFile_ntdll",
            || {
                crate::crash_guard::with_reentrancy_guard(
                    || {
                        // Classification logic
                        if let Some(_deny) = classify_and_log_path(&path, "CREATE", "NtCreateFile") {
                            return crate::fail_closed!(StatusAccessDenied);
                        }
                        // Call original via retour trampoline
                        let original_ptr = crate::ntdll_patcher::get_original_trampoline("NtCreateFile");
                        if original_ptr.is_none() {
                            // Fallback: resolve directly from ntdll
                            let fallback = crate::resolve_nt_create_file().unwrap_or_else(|| {
                                panic!("NtCreateFile original unavailable and resolution failed")
                            });
                            return fallback(filehandle, desiredaccess, ...);
                        }
                        let original: NtCreateFileFn = std::mem::transmute(original_ptr.unwrap());
                        original(filehandle, desiredaccess, objectattributes, ...)
                    },
                    || {
                        // Reentrancy fallback: call original without classification
                        let original_ptr = crate::ntdll_patcher::get_original_trampoline("NtCreateFile");
                        if let Some(ptr) = original_ptr {
                            let original: NtCreateFileFn = std::mem::transmute(ptr);
                            original(filehandle, desiredaccess, objectattributes, ...)
                        } else {
                            let fallback = crate::resolve_nt_create_file().unwrap_or_else(|| {
                                panic!("NtCreateFile original unavailable and resolution failed")
                            });
                            fallback(filehandle, desiredaccess, ...)
                        }
                    },
                )
            },
            || {
                // Panic fallback
                let fallback = crate::resolve_nt_create_file().unwrap_or_else(|| {
                    panic!("NtCreateFile original unavailable and resolution failed")
                });
                fallback(filehandle, desiredaccess, objectattributes, ...)
            },
        )
    }
    ```
