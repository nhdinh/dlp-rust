       /// # Behavior
       /// - `None` (no LDAP config) returns `None`.
       /// - In-range values pass through.
       /// - Out-of-range values are clamped to [60, 3600] and a `warn!` is emitted.
       pub fn resolved_cache_ttl(&self) -> Option<std::time::Duration> {

           let clamped = raw.clamp(LDAP_CACHE_TTL_MIN_SECS, LDAP_CACHE_TTL_MAX_SECS);
           if clamped != raw {
               warn!(

                   applied = clamped,
                   min = LDAP_CACHE_TTL_MIN_SECS,
                   max = LDAP_CACHE_TTL_MAX_SECS,

    2. **Add a shutdown token/cancellation mechanism**:
       Replace the ad-hoc `drop(event_loop_handle)` and individual `send(true)` calls with a coordinated shutdown using `tokio::sync::broadcast` or `tokio_util::sync::CancellationToken`. However, to minimize disruption to existing code, enhance the existing `watch::channel<bool>` pattern:
       - Add a `shutdown_tx: tokio::sync::watch::Sender<bool>` that is cloned and passed to all subsystems that need cancellation.
       - For the disk enumeration task (which is spawned via `runtime_handle.spawn` and is NOT a tokio task within the runtime), use an `Arc<AtomicBool>` or a `tokio::sync::watch::Receiver<bool>` passed into `spawn_disk_enumeration_task`.

       Simpler approach: wrap the existing shutdown sequence in a `tokio::time::timeout`:
       ```rust
       let shutdown_result = tokio::time::timeout(SHUTDOWN_TIMEOUT, async {
           // existing shutdown sequence
       }).await;
       if shutdown_result.is_err() {
           tracing::error!("graceful shutdown exceeded {}s timeout -- force-terminating", SHUTDOWN_TIMEOUT.as_secs());
       }
       ```

    3. **Cancel in-flight disk enumeration within 5s**:
       The disk enumeration task is spawned via `runtime_handle.spawn(async move { ... })` in `spawn_disk_enumeration_task`. To make it cancellable:

       - Inside the enumeration loop, check `shutdown_rx.has_changed()` or use `tokio::select!` with a timeout.
       - In `run_loop`, create a `disk_shutdown_tx`/`disk_shutdown_rx` pair and pass `disk_shutdown_rx` to `spawn_disk_enumeration_task`.
