```rust
/// Enriches an AuditEvent with source/destination application identity.
///
/// Resolves the process image path from `pid` via `get_application_metadata`,
/// then sets `source_application` on the event. If resolution fails,
/// uses the AGENT-UNKNOWN sentinel.
///
/// # Arguments
///
/// * `event` — the audit event to enrich (mutated in place)
/// * `pid` — the process ID of the operation initiator
pub fn enrich_audit_with_app_identity(event: &mut dlp_common::AuditEvent, pid: u32) {
    let (app_path, _app_hash) = get_application_metadata(pid);
    let app_identity = app_path.map(|path| {
        dlp_common::AppIdentity {
            name: std::path::Path::new(&path)
                .file_stem()
                .and_then(|s| s.to_str())
                .unwrap_or("UNKNOWN")
                .to_string(),
            path: Some(path),
            aumid: None, // Win32 processes have no AUMID
        }
    }).unwrap_or_else(dlp_common::endpoint::agent_unknown_app);
    event.source_application = Some(app_identity);
    // destination_application is context-dependent; leave for caller
}
```

Also add a helper for destination application:
```rust
/// Sets destination_application on an AuditEvent.
///
/// Used when the destination is known (e.g., copy to a specific path,
/// paste to a specific application). If None, sets AGENT-UNKNOWN.
pub fn set_destination_application(
    event: &mut dlp_common::AuditEvent,
    dest: Option<dlp_common::AppIdentity>,
) {
    event.destination_application = Some(dest.unwrap_or_else(dlp_common::endpoint::agent_unknown_app));
}
```
