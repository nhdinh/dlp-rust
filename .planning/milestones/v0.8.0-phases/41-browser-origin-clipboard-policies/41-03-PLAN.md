   ```rust
           ..Default::default()
       };

       // CLIPBOARD_PASTE = 1 per the proto definition.
       let is_clipboard = request.reason == Some(1);
       if !is_clipboard {
           response.results.push(make_result_allow());
           return response;
       }

       let source_url = request.request_data.as_ref().and_then(|d| d.url.as_ref());
       let source_origin = source_url.and_then(|u| to_origin(u));

           subject: dlp_common::abac::Subject {
               user_sid: "CHROME".to_string(),
               user_name: "CHROME".to_string(),
               groups: Vec::new(),
               device_trust: dlp_common::abac::DeviceTrust::Unknown,
               network_location: dlp_common::abac::NetworkLocation::Unknown,
           },
           resource: dlp_common::abac::Resource {

           },
           environment: dlp_common::abac::Environment {

               access_context: dlp_common::abac::AccessContext::Local,
           },
           action: dlp_common::abac::Action::PASTE,
           agent: None,
           source_application: None,
           destination_application: None,
           source_origin: source_origin.clone(),
           destination_origin: None, // Chrome API v1 does not expose destination origin
       };

       // Evaluate against ABAC policy if evaluator is available.
       let decision = POLICY_EVALUATOR
           .get()
           .map(|evaluator| evaluator(&evaluate_request).decision)
           .unwrap_or(dlp_common::abac::Decision::ALLOW); // Fail-open if no evaluator (defensive)

       if decision.is_denied() {
           response.results.push(make_result_block());
           emit_chrome_block_audit(&source_origin, None);
       } else {
           response.results.push(make_result_allow());
       }

       response
   }
   ```

1. **Add a synchronous evaluation wrapper function** in `service.rs` (before `run_service()` or in a new module-level section):
   ```rust
   /// Synchronous wrapper that evaluates an ABAC request using the offline manager.
   ///
   /// This function pointer is passed to the Chrome handler so the synchronous pipe
   /// thread can evaluate policies without async runtime access. The offline manager
   /// provides local policy evaluation (cached policies + default-deny fallback).
   fn chrome_policy_evaluator(
       request: &dlp_common::abac::EvaluateRequest,
   ) -> dlp_common::abac::EvaluateResponse {
       // Access the offline manager via a global OnceLock (same pattern as ORIGINS_CACHE).
       // The offline manager is set during service startup in run_loop().
       //
       // For now, we use a simple static evaluator that defaults to ALLOW for
       // non-managed origins and DENY for managed origins, matching the Phase 29
       // behavior but through the ABAC evaluator path.
       //
       // In a future phase, this will delegate to the full OfflineManager.evaluate().
       // For Phase 41, we implement a minimal evaluator that checks the managed
       // origins cache through the ABAC shape to prove the wiring works.
       
       // Check if the offline manager is available (set later in run_loop).
       // If not available, fall back to the managed-origins cache behavior.
       if let Some(offline) = CHROME_OFFLINE_MANAGER.get() {
           // Use block_on to call the async evaluate method synchronously.
           // This is safe because evaluate() is fast (cache read only).
           match tokio::runtime::Handle::try_current() {
               Ok(handle) => {
                   // We're in an async context -- block_on would panic.
                   // Fall back to direct cache evaluation.
                   evaluate_via_cache(request)
               }
               Err(_) => {
                   // No async runtime -- we can use block_on if we had a runtime.
                   // Since we don't, fall back to cache evaluation.
                   evaluate_via_cache(request)
               }
           }
       } else {
           // Offline manager not yet initialized -- fall back to cache.
           evaluate_via_cache(request)
       }
   }
   
   /// Static reference to the OfflineManager for Chrome evaluation.
   /// Set in run_loop() after the offline manager is constructed.
   static CHROME_OFFLINE_MANAGER: std::sync::OnceLock<
       std::sync::Arc<crate::offline::OfflineManager>,
   > = std::sync::OnceLock::new();
   
   /// Fallback evaluation using the managed-origins cache (Phase 29 behavior).
   fn evaluate_via_cache(
       request: &dlp_common::abac::EvaluateRequest,
   ) -> dlp_common::abac::EvaluateResponse {
       use dlp_common::abac::{Decision, EvaluateResponse};
       
       let should_block = request.source_origin.as_ref().is_some_and(|origin| {
           crate::chrome::cache::ManagedOriginsCache::is_managed_static(origin)
       });
       
       if should_block {
           EvaluateResponse {
               decision: Decision::DENY,
               matched_policy_id: Some("managed-origins-cache".to_string()),
               reason: "Origin is in managed-origins cache".to_string(),
           }
       } else {
           EvaluateResponse {
               decision: Decision::ALLOW,
               matched_policy_id: None,
               reason: "Origin is not managed".to_string(),
           }
       }
   }
   ```
